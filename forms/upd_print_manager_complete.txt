Функция ПолныйПутьКМакету(ДатаДокумента, ЭтоУПД, ЕстьПрослеживаемыеТовары = Ложь) Экспорт

	НачалоИспользованияСФ981 = Константы.НачалоИспользованияСФ981.Получить();
	НачалоИспользованияСФ534 = '20210701';
	НачалоИспользованияСФ2023Кв4 = '20231001';
	НачалоИспользованияСФ1096 = '20241001';
	
	Если ДатаДокумента = Неопределено Тогда

		ДатаДокумента = НачалоИспользованияСФ534;

	КонецЕсли;

	Тело = "Обработка.ПечатьСчетФактура.";

	Если ДатаДокумента < НачалоИспользованияСФ981 Тогда

		ПолныйПутьКМакету = Тело + ?(ЭтоУПД, "ПФ_MXL_УниверсальныйПередаточныйДокумент", "ПФ_MXL_СчетФактура1137");

	ИначеЕсли ДатаДокумента < НачалоИспользованияСФ534 Тогда
		
		ПолныйПутьКМакету = Тело + ?(ЭтоУПД, "ПФ_MXL_УниверсальныйПередаточныйДокумент981", "ПФ_MXL_СчетФактура981");
		
	ИначеЕсли ДатаДокумента < НачалоИспользованияСФ1096 Тогда
			
		Если ЕстьПрослеживаемыеТовары 
			И ДатаДокумента >= НачалоИспользованияСФ2023Кв4 Тогда   
			ПолныйПутьКМакету = Тело
			+ ?(ЭтоУПД, "ПФ_MXL_УниверсальныйПередаточныйДокумент2023Кв4Прослеж", "ПФ_MXL_СчетФактура2023Кв4Прослеж");
		Иначе
			ПолныйПутьКМакету = Тело
				+ ?(ЭтоУПД, "ПФ_MXL_УниверсальныйПередаточныйДокумент534", "ПФ_MXL_СчетФактура534")
				+ ?(ЕстьПрослеживаемыеТовары, "Прослеж", "");
		КонецЕсли;		

	Иначе
		
		ПолныйПутьКМакету = Тело
			+ ?(ЭтоУПД, "ПФ_MXL_УниверсальныйПередаточныйДокумент1096", "ПФ_MXL_СчетФактура1096")
			+ ?(ЕстьПрослеживаемыеТовары, "Прослеж", "");
	КонецЕсли;

	Возврат ПолныйПутьКМакету;

КонецФункции

Функция СформироватьПФ(ОписаниеПечатнойФормы, ДанныеОбъектовПечати, ОбъектыПечати, ЭтоУПД) Экспорт
	Перем Ошибки;

	ТабличныйДокумент = ОписаниеПечатнойФормы.ТабличныйДокумент;
	ДанныеПечати = Новый Структура;

	ЕстьКолонкаЭтоКорректировка = (ДанныеОбъектовПечати.Колонки.Найти("ЭтоКорректировка") <> Неопределено);
	ЕстьКолонкаАкциз = (ДанныеОбъектовПечати.Колонки.Найти("ЕстьКолонкаАкциз") <> Неопределено);
	ЕстьКолонкаСопроводительныеДокументы =
		(ДанныеОбъектовПечати.Колонки.Найти("СопроводительныеДокументы") <> Неопределено);
	ЕстьКолонкаДанныеОТранспортировке = (ДанныеОбъектовПечати.Колонки.Найти("ДанныеОТранспортировке") <> Неопределено);
	ТекстСообщенияМакетИзменен = ПечатьДокументовУНФ.ТекстСообщенияМакетИзменен(НСтр("ru = 'Счета-фактуры'"));
	Для Каждого ДанныеОбъекта Из ДанныеОбъектовПечати Цикл
		
		Если ТипЗнч(ДанныеОбъекта.Ссылка) =  Тип("ДокументСсылка.КорректировкаПоступления")
			И ЗначениеЗаполнено(ДанныеОбъекта.ДатаИсправления) И ЗначениеЗаполнено(ДанныеОбъекта.НомерИсправления) Тогда
			
			ДатаДокументаОригинал = ДанныеОбъекта.ДатаДокумента;
			НомерОригинал = ДанныеОбъекта.Номер;
			
			ДанныеОбъекта.ДатаДокумента = ДанныеОбъекта.ДатаИсправления;
			ДанныеОбъекта.Номер = ДанныеОбъекта.НомерИсправления;
			ДанныеОбъекта.НомерДляПечати = ДанныеОбъекта.Номер;
			
			ДанныеОбъекта.ДатаИсправления = ДатаДокументаОригинал;
			ДанныеОбъекта.НомерИсправления = НомерОригинал;
			
		КонецЕсли;
		
		ПолныйПутьКМакетуПечатнойФормы = ПолныйПутьКМакету(ДанныеОбъекта.ДатаДокумента, ЭтоУПД, ДанныеОбъекта.ЕстьПрослеживаемыеТовары);
		Если ЗначениеЗаполнено(ОписаниеПечатнойФормы.ПолныйПутьКМакету)
			 И ОписаниеПечатнойФормы.ПолныйПутьКМакету <> ПолныйПутьКМакетуПечатнойФормы Тогда

			ВызватьИсключение НСтр("ru ='Нельзя пакетно формировать счета-фактуры разных постановлений.'");

		ИначеЕсли Не ЗначениеЗаполнено(ОписаниеПечатнойФормы.ПолныйПутьКМакету) Тогда

			ОписаниеПечатнойФормы.ПолныйПутьКМакету = ПолныйПутьКМакетуПечатнойФормы;

		КонецЕсли;

		Макет = УправлениеПечатью.МакетПечатнойФормы(ОписаниеПечатнойФормы.ПолныйПутьКМакету);

		ОбластиМакета = Новый Структура;
		ОбластиМакета.Вставить("ОбластьЗаголовокТаблицы", ПечатьДокументовУНФ.ПолучитьОбластьБезопасно(Макет,
			"ЗаголовокТаблицы", , Ошибки));
		ОбластиМакета.Вставить("ОбластьСтроки", ПечатьДокументовУНФ.ПолучитьОбластьБезопасно(Макет, "Строка", , Ошибки));
		ОбластиМакета.Вставить("ИтогоПоСтранице", ПечатьДокументовУНФ.ПолучитьОбластьБезопасно(Макет,
			"ИтогоПоСтранице", , Ошибки));
		ОбластиМакета.Вставить("ОбластьИтого", ПечатьДокументовУНФ.ПолучитьОбластьБезопасно(Макет, "Итого", , Ошибки));
		ОбластиМакета.Вставить("ОбластьНумерацияЛистов", ПечатьДокументовУНФ.ПолучитьОбластьБезопасно(Макет,
			"НумерацияЛистов", , Ошибки));

		ИмяОбластиПодвал = "Подвал";
		ИмяОбласти = ?(ЭтоУПД, "ПодвалСФаксимиле", ИмяОбластиПодвал);
		ОбластиМакета.Вставить("ОбластьПодвалСФаксимиле", ПечатьДокументовУНФ.ПолучитьОбластьБезопасно(Макет,
			ИмяОбласти, "", Ошибки));
		ОбластиМакета.Вставить("ОбластьПодвалБезФаксимиле", ПечатьДокументовУНФ.ПолучитьОбластьБезопасно(Макет,
			ИмяОбластиПодвал, "", Ошибки));
		ОбластиМакета.Вставить("ОбластьПодвал", ПечатьДокументовУНФ.ПолучитьОбластьБезопасно(Макет,
			ИмяОбластиПодвал, "", Ошибки));

		Если ЭтоУПД Тогда
			ИмяОбласти = "ПодвалНакладнойСФаксимиле";
			ОбластиМакета.Вставить("ОбластьПодвалНакладнойСФаксимиле", ПечатьДокументовУНФ.ПолучитьОбластьБезопасно(
				Макет, ИмяОбласти, "", Ошибки));
			ИмяОбласти = "ПодвалНакладной";
			ОбластиМакета.Вставить("ОбластьПодвалНакладной", ПечатьДокументовУНФ.ПолучитьОбластьБезопасно(Макет,
				ИмяОбласти, "", Ошибки));
		КонецЕсли;
		
		СтавкаНДС0 = УправлениеНебольшойФирмойПовтИсп.ПолучитьСтавкуНДСНоль();
		ДокументФормируетсяСогласноФЗ56 = ПечатьДокументовУНФ.ДокументФормируетсяСогласноФЗ56(
			ДанныеОбъекта.ДатаДокумента);
		ДокументФормируетсяСогласноСФ981 = (ДанныеОбъекта.ДатаДокумента
											>= Константы.НачалоИспользованияСФ981.Получить());
		ИспользуетсяПостановление1137 = (ДанныеОбъекта.ДатаДокумента >= Константы.НачалоИспользованияСФ1137.Получить());
		НомерПостановления1137 = ПечатьДокументовУНФ.ВерсияПостановленияНДС1137(ДанныеОбъекта.ДатаДокумента, ДанныеОбъекта.ЕстьПрослеживаемыеТовары);
		НеобходимоВНоменклатуруДобавитьКодыТНВЭД = (ДанныеОбъекта.ДатаДокумента >= Дата(2016, 07, 01));
		
		СвойстваПостановления = Документы.СчетФактура.ОписаниеПечатнойФормыСчетаФактуры(НомерПостановления1137);
		СвойстваПостановления.ЗаполнятьДокументыОтгрузки = (ДанныеОбъекта.ВидОперации = Перечисления.ВидыОперацийСчетФактура.Продажа
			ИЛИ ДанныеОбъекта.ВидОперации = Перечисления.ВидыОперацийСчетФактураПолученный.Поступление);
			
		// Прослеживаемость
		ВыводитьРНПТ = СвойстваПостановления.ЕстьПрослеживаемость
			И ДанныеОбъекта.ТаблицаЗапасы.Колонки.Найти("СведенияОПрослеживаемости") <> Неопределено
			И ЗначениеЗаполнено(Макет.Области.Найти("ОбластьПрослеживаемость"));
		// Конец Прослеживаемость

		ЕстьЗапасы = ЕстьЗапасы(ДанныеОбъекта);
		ЕстьНаборы = ДанныеОбъекта.ТаблицаЗапасы.Колонки.Найти("НоменклатураНабора") <> Неопределено;
		
		// Заголовок
		ИмяОбласти = ?(ДокументФормируетсяСогласноФЗ56, "ЗаголовокПФ625", "ЗаголовокПФ");
		ОбластьМакета = ПечатьДокументовУНФ.ПолучитьОбластьБезопасно(Макет, ИмяОбласти, , Ошибки);
		Если ОбластьМакета = Неопределено Тогда

			ВызватьИсключение ТекстСообщенияМакетИзменен;

		Иначе

			ТабличныйДокумент.Вывести(ОбластьМакета);

		КонецЕсли;

		
		// Шапка
		СведенияОПоставщике = ПечатьДокументовУНФ.СведенияОЮрФизЛице(ДанныеОбъекта.Организация,
			ДанныеОбъекта.ДатаДокумента, , );
		СведенияОПокупателе = ПечатьДокументовУНФ.СведенияОЮрФизЛице(ДанныеОбъекта.Контрагент,
			ДанныеОбъекта.ДатаДокумента, , );

		СведенияОГрузоотправителе = ПечатьДокументовУНФ.СведенияОЮрФизЛице(ДанныеОбъекта.Грузоотправитель,
			ДанныеОбъекта.ДатаДокумента, , );
		СведенияОГрузополучателе = ПечатьДокументовУНФ.СведенияОЮрФизЛице(ДанныеОбъекта.Грузополучатель,
			ДанныеОбъекта.ДатаДокумента, , );

		ДанныеПечати.Вставить("ВидОперации", ДанныеОбъекта.ВидОперации);
		Если ДанныеОбъекта.Владелец().Колонки.Найти("НомерДляПечати") = Неопределено Тогда
			ДанныеПечати.Вставить("Номер", ПечатьДокументовУНФ.ПолучитьНомерНаПечатьСУчетомДатыДокумента(
				ДанныеОбъекта.ДатаДокумента, ДанныеОбъекта.Номер, ДанныеОбъекта.Префикс));
		Иначе
			ДанныеПечати.Вставить("Номер", ДанныеОбъекта.НомерДляПечати);
		КонецЕсли;

		Если Не ПустаяСтрока(ДанныеОбъекта.ЦифровойИндексОбособленногоПодразделения) Тогда

			ДанныеПечати.Номер = ДанныеПечати.Номер + "/" + СокрЛП(
				ДанныеОбъекта.ЦифровойИндексОбособленногоПодразделения);

		КонецЕсли;

		ДанныеПечати.Вставить("Дата", Формат(ДанныеОбъекта.ДатаДокумента, "ДФ='дд ММММ гггг ""г.""'"));
		ДанныеПечати.Вставить("НомерИсправления", ДанныеОбъекта.НомерИсправления);
		ДанныеПечати.Вставить("ДатаИсправления", ?(ЗначениеЗаполнено(ДанныеОбъекта.ДатаИсправления), Формат(
			ДанныеОбъекта.ДатаИсправления, "ДФ='дд ММММ гггг ""г.""'"), Неопределено));
		ДанныеПечати.Вставить("СтатусУПД", ДанныеОбъекта.СтатусУПД);

		
		Если СтрНайти("" + ДанныеПечати.Дата, " г.") > 0 Тогда
			ТекстШаблона = НСтр("ru ='%1 № %2 от %3'");
		Иначе
			ТекстШаблона = НСтр("ru ='%1 № %2 от %3 г.'");
		КонецЕсли;
		ТекстНаименование = ?(ЭтоУПД, НСтр("ru ='Универсальный передаточный документ'"), НСтр("ru ='Счет-фактура'"));
		ДанныеПечати.Вставить("ПредставлениеДокумента",
			СтрШаблон(ТекстШаблона, ТекстНаименование, ДанныеПечати.Номер, ДанныеПечати.Дата));
		
		Если ТипЗнч(ДанныеОбъекта.Организация) = Тип("СправочникСсылка.Организации") Тогда
			НаименованиеДляПечатиПоставщик = "НаименованиеДляПечати,";
		Иначе
			НаименованиеДляПечатиПоставщик = "ПолноеНаименование,";
		КонецЕсли;
		
		Если ТипЗнч(ДанныеОбъекта.Контрагент) = Тип("СправочникСсылка.Организации") Тогда
			НаименованиеДляПечатиПокупатель = "НаименованиеДляПечати,";
		Иначе
			НаименованиеДляПечатиПокупатель = "ПолноеНаименование,";
		КонецЕсли;
		
		ДанныеПечати.Вставить("ПредставлениеПоставщика", ПечатьДокументовУНФ.ОписаниеОрганизации(
			СведенияОПоставщике, НаименованиеДляПечатиПоставщик));
		ДанныеПечати.Вставить("АдресПоставщика", ПечатьДокументовУНФ.ОписаниеОрганизации(
			СведенияОПоставщике, "ЮридическийАдрес,"));

		ДанныеПечати.Вставить("ПредставлениеПокупателя", ПечатьДокументовУНФ.ОписаниеОрганизации(
			СведенияОПокупателе, НаименованиеДляПечатиПокупатель));
		ДанныеПечати.Вставить("АдресПокупателя", ПечатьДокументовУНФ.ОписаниеОрганизации(
			СведенияОПокупателе, "ЮридическийАдрес,"));

		ПредставлениеГрузоотправителя = "";
		ПредставлениеГрузополучателя = "";
		Если ЕстьЗапасы Тогда

			Если ДанныеОбъекта.Грузоотправитель = ДанныеОбъекта.Организация Тогда

				ПредставлениеГрузоотправителя = НСтр("ru = 'он же'");

			ИначеЕсли ЗначениеЗаполнено(ДанныеОбъекта.Грузоотправитель) Тогда

				ПредставлениеГрузоотправителя = ПечатьДокументовУНФ.ОписаниеОрганизации(
					СведенияОГрузоотправителе, НаименованиеДляПечатиПоставщик + "ФактическийАдрес,");
				
			КонецЕсли;

			АдресДоставки = ДанныеОбъекта.АдресДоставки;
			ПредставлениеГрузополучателя = ?(ПустаяСтрока(АдресДоставки),
				ПечатьДокументовУНФ.ОписаниеОрганизации(СведенияОГрузополучателе,
				НаименованиеДляПечатиПокупатель + "ФактическийАдрес,"), ПечатьДокументовУНФ.ОписаниеОрганизации(
				СведенияОГрузополучателе, НаименованиеДляПечатиПокупатель) + "; " + АдресДоставки);

		КонецЕсли;
		ДанныеПечати.Вставить("ПредставлениеГрузоотправителя", ПредставлениеГрузоотправителя);
		ДанныеПечати.Вставить("ПредставлениеГрузополучателя", ПредставлениеГрузополучателя);

		ИННПоставщика = ПечатьДокументовУНФ.ОписаниеОрганизации(СведенияОПоставщике, "ИНН,", Ложь);

		Если ДанныеОбъекта.ОбособленноеПодразделениеПоставщика = ДанныеОбъекта.Организация Тогда

			КПППоставщика = ПечатьДокументовУНФ.ОписаниеОрганизации(СведенияОПоставщике, "КПП,", Ложь);

		Иначе

			КПППоставщика = ОбщегоНазначения.ЗначениеРеквизитаОбъекта(
				ДанныеОбъекта.ОбособленноеПодразделениеПоставщика, "КПП");

		КонецЕсли;

		ИННПоставщика = ИННПоставщика + ?(ПустаяСтрока(КПППоставщика), "", "/" + КПППоставщика);
		ДанныеПечати.Вставить("ИННПоставщика", ИННПоставщика);

		ИННПокупателя = ПечатьДокументовУНФ.ОписаниеОрганизации(СведенияОПокупателе, "ИНН,", Ложь);

		Если ДанныеОбъекта.ОбособленноеПодразделениеПокупателя = ДанныеОбъекта.Контрагент Тогда

			КПППокупателя = ПечатьДокументовУНФ.ОписаниеОрганизации(СведенияОПокупателе, "КПП,", Ложь);

		Иначе

			КПППокупателя = ОбщегоНазначения.ЗначениеРеквизитаОбъекта(
				ДанныеОбъекта.ОбособленноеПодразделениеПокупателя, "КПП");

		КонецЕсли;

		ИННПокупателя = ИННПокупателя + ?(ПустаяСтрока(КПППокупателя), "", "/" + КПППокупателя);
		ДанныеПечати.Вставить("ИННПокупателя", ИННПокупателя);
		
		// ПоДокументу
		ПредставлениеПлатежныхДокументов = "";
		Если ТипЗнч(ДанныеОбъекта.ПлатежныеДокументы) = Тип("ТаблицаЗначений") Тогда

			ШаблонПредставления = НСтр("ru ='№ %1 от %2'");
			Для Каждого ПлатежныйДокумент Из ДанныеОбъекта.ПлатежныеДокументы Цикл

				ПредставлениеПлатежныхДокументов = ПредставлениеПлатежныхДокументов + ?(ПустаяСтрока(
					ПредставлениеПлатежныхДокументов), "", ", ") + СтрШаблон(ШаблонПредставления,
					ПлатежныйДокумент.НомерПлатежноРасчетногоДокумента, Формат(
					ПлатежныйДокумент.ДатаПлатежноРасчетногоДокумента, "ДФ=dd.MM.yyyy"));

			КонецЦикла;

		КонецЕсли;
		ДанныеПечати.Вставить("ПоДокументу", ПредставлениеПлатежныхДокументов);
		
		// ДокументыОбОтгрузке
		Если СвойстваПостановления.ЕстьДокументыОтгрузки Тогда
			ДополнитьДанныеШапкиСведениямиОДокументеОтгрузки(ДанныеПечати, ДанныеОбъекта.ТаблицаЗапасы, ЭтоУПД, СвойстваПостановления);
		КонецЕсли;
		
		// без перехода права собственности
		Если ЭтоУПД И ТипЗнч(ДанныеОбъекта.Ссылка) = Тип("ДокументСсылка.РасходнаяНакладная")Тогда
			// В соответствии с письмом ФНС России от 21.10.2013 N ММВ-20-3/96@ "Об отсутствии налоговых рисков при применении 
			// налогоплательщиками первичного документа, составленного на основе формы счета-фактуры", в УПД со статусом 2 при передаче
			// без перехода права собственности строки (2),(2а),(2б) и (6),(6а),(6б) не заполняются.
			
			БезПереходаПраваСобственности = Ложь;
			ЕстьОснованиеПечатиСсылка = ДанныеОбъекта.Владелец().Колонки.Найти("ОснованиеПечатиСсылка") <> Неопределено;
			Если ЕстьОснованиеПечатиСсылка Тогда  
				Если  Тип("СправочникСсылка.ДоговорыКонтрагентов") = ТипЗнч(ДанныеОбъекта.ОснованиеПечатиСсылка) Тогда 				
					БезПереходаПраваСобственности = ДанныеОбъекта.ОснованиеПечатиСсылка.ВидДоговора = Перечисления.ВидыДоговоров.СКомиссионером;
				КонецЕсли; 			
			КонецЕсли;				
			
			Если БезПереходаПраваСобственности И ДанныеОбъекта.СтатусУПД = 2 Тогда
				
				// Поставщик и Грузоотправитель
				Если ТипЗнч(ДанныеПечати.ПредставлениеГрузоотправителя) = Тип("Строка") Тогда // "он же"
					ДанныеПечати.ПредставлениеГрузоотправителя = ДанныеПечати.ПредставлениеПоставщика 
					+ ?(ПустаяСтрока(ДанныеПечати.АдресПоставщика), "", ", ") + ДанныеПечати.АдресПоставщика;
				КонецЕсли;
				
				ДанныеПечати.ПредставлениеПоставщика = "--";         // (2)
				ДанныеПечати.АдресПоставщика         = "--";         // (2а)
				ДанныеПечати.ИННПоставщика           = "--";         //
				
				// Покупатель и Грузополучатель
				Если ТипЗнч(ДанныеПечати.ПредставлениеГрузополучателя) = Тип("Строка") Тогда // "он же"
					ДанныеПечати.ПредставлениеГрузополучателя = ДанныеПечати.ПредставлениеПокупателя
					+ ?(ПустаяСтрока(ДанныеПечати.АдресПокупателя), "", ", ") + ДанныеПечати.АдресПокупателя;
				КонецЕсли;
				
				ДанныеПечати.ПредставлениеПокупателя = "--";         // (6)
				ДанныеПечати.АдресПокупателя         = "--";         // (6а)
				ДанныеПечати.ИННПокупателя           = "--";         //
				
			КонецЕсли;
		КонецЕсли;
		// 
		
		ДанныеПечати.Вставить("Валюта", СокрЛП(ДанныеОбъекта.ВалютаНаименование) + ", " + СокрЛП(
			ДанныеОбъекта.ВалютаКод));
		ДанныеПечати.Вставить("ИдентификаторГосКонтракта", ДанныеОбъекта.ИдентификаторГосКонтракта);

		УчестьРазличияСводногоСФ(ДанныеОбъекта, ДанныеПечати, ЭтоУПД);
		ОбработатьПустыеПоляШапки(ДанныеПечати, ЕстьЗапасы, ЭтоУПД);

		ОбластьШапка = ПечатьДокументовУНФ.ПолучитьОбластьБезопасно(Макет, "Шапка", , Ошибки);
		Если ОбластьШапка <> Неопределено Тогда

			ОбластьШапка.Параметры.Заполнить(ДанныеПечати);
			ШтрихкодированиеПечатныхФорм.ВывестиШтрихкодВТабличныйДокумент(ТабличныйДокумент, Макет, ОбластьШапка,
				ДанныеОбъекта.Ссылка);
			ТабличныйДокумент.Вывести(ОбластьШапка);

		КонецЕсли;

		Если ДокументФормируетсяСогласноФЗ56 Тогда

			ОбластьМакета = ПечатьДокументовУНФ.ПолучитьОбластьБезопасно(Макет, "ИдентификаторГосКонтракта", , Ошибки);
			Если ОбластьМакета <> Неопределено Тогда

				ОбластьМакета.Параметры.Заполнить(ДанныеПечати);
				ТабличныйДокумент.Вывести(ОбластьМакета);

			КонецЕсли;

		КонецЕсли;
		
		// Табличная часть
		Если ОбластиМакета.ОбластьЗаголовокТаблицы <> Неопределено Тогда

			ТабличныйДокумент.Вывести(ОбластиМакета.ОбластьЗаголовокТаблицы);

		КонецЕсли;

		СтруктураИтогов = Новый Структура;
		СтруктураИтогов.Вставить("НомерЛиста", 1);
		СтруктураИтогов.Вставить("НПП", 0);
		СтруктураИтогов.Вставить("КоличествоСтрок", ДанныеОбъекта.ТаблицаЗапасы.Количество());
		СтруктураИтогов.Вставить("ПоСтраницеСтоимость", 0);
		СтруктураИтогов.Вставить("ПоСтраницеСуммаНДС", 0);
		СтруктураИтогов.Вставить("ПоСтраницеВсего", 0);
		СтруктураИтогов.Вставить("Стоимость", 0);
		СтруктураИтогов.Вставить("СуммаНДС", 0);
		СтруктураИтогов.Вставить("Всего", 0);
		СтруктураИтогов.Вставить("ПредставлениеДокумента", ДанныеПечати.ПредставлениеДокумента);
		СтруктураИтогов.Вставить("ЕстьНаборы", ЕстьНаборы);
		СтруктураИтогов.Вставить("ЭтоУПД", ЭтоУПД);
		СтруктураИтогов.Вставить("ЕстьСтавкаНольПроцентов", Ложь);

		Если ОбластиМакета.ОбластьСтроки <> Неопределено Тогда
			
			КартаКоличествЛотковУПД = Неопределено;
			Если ЭтоУПД Или ДанныеОбъекта.ВидОперации = Перечисления.ВидыОперацийСчетФактура.Продажа Тогда
				КартаКоличествЛотковУПД = УПДЛоткиПодготовитьКарту(ДанныеОбъекта, ОбластиМакета.ОбластьСтроки);
			КонецЕсли;

			ПараметрыНоменклатуры = Новый Структура;
			
			КоличествоСтрок = ДанныеОбъекта.ТаблицаЗапасы.Количество();
			Для Каждого СтрокаТабличнойЧасти Из ДанныеОбъекта.ТаблицаЗапасы Цикл
				
				ОбластьМакета = ОбластиМакета.ОбластьСтроки;
				Если СвойстваПостановления.ЕстьПрослеживаемость Тогда
					ОбластьЗапись = ОбластьМакета.Область("ОбластьЗапись");
				КонецЕсли;
				
				ДанныеПечати.Очистить();
				ПараметрыНоменклатуры.Очистить();

				Если СтруктураИтогов.ЕстьНаборы И СтрокаТабличнойЧасти.ЭтоНабор = Истина Тогда
					НомерСтроки = 0;
				Иначе
					СтруктураИтогов.НПП = СтруктураИтогов.НПП + 1;
					НомерСтроки = СтруктураИтогов.НПП;
				КонецЕсли;
				ДанныеПечати.Вставить("НомерСтроки", НомерСтроки);

				ПараметрыНоменклатуры.Вставить("Содержание", СтрокаТабличнойЧасти.Содержание);
				ПараметрыНоменклатуры.Вставить("ПредставлениеНоменклатуры",
					СтрокаТабличнойЧасти.ПредставлениеНоменклатуры);
				ПараметрыНоменклатуры.Вставить("ПредставлениеХарактеристики", СтрокаТабличнойЧасти.Характеристика);
				ПараметрыНоменклатуры.Вставить("НеобходимоВыделитьКакСоставНабора", ?(СтруктураИтогов.ЕстьНаборы,
					СтрокаТабличнойЧасти.НеобходимоВыделитьКакСоставНабора, Ложь));

				Если ДокументФормируетсяСогласноСФ981 Тогда

					ДанныеПечати.Вставить("ТоварКодТНВЭД", Неопределено);
					Если СтрокаТабличнойЧасти.СтавкаНДС = СтавкаНДС0 Тогда

						ДанныеПечати.ТоварКодТНВЭД = СокрЛП(СтрокаТабличнойЧасти.КодТНВЭД);

					КонецЕсли;

				ИначеЕсли НеобходимоВНоменклатуруДобавитьКодыТНВЭД Тогда

					ПараметрыНоменклатуры.Вставить("КодТНВЭД", СтрокаТабличнойЧасти.КодТНВЭД);

				КонецЕсли;

				ДанныеПечати.Вставить("ТоварНаименование", ПечатьДокументовУНФ.ПредставлениеНоменклатуры(
					ПараметрыНоменклатуры));
				ДанныеПечати.Вставить("ПредставлениеКодаНоменклатуры",
					ПечатьДокументовУНФ.ПредставлениеКодаНоменклатуры(СтрокаТабличнойЧасти));
				ДанныеПечати.Вставить("ЭтоНабор", ?(СтруктураИтогов.ЕстьНаборы, СтрокаТабличнойЧасти.ЭтоНабор = Истина,
					Ложь));
				ДанныеПечати.Вставить("ЕдиницаИзмеренияКод", СтрокаТабличнойЧасти.ЕдиницаИзмеренияПоОКЕИ_Код);
				ДанныеПечати.Вставить("ЕдиницаИзмерения", СтрокаТабличнойЧасти.ЕдиницаИзмеренияПоОКЕИ_Наименование);

				Если ТипЗнч(СтрокаТабличнойЧасти.Количество) = Тип("Число") И ТипЗнч(
					СтрокаТабличнойЧасти.КоэффициентЕдиницыИзмерения) = Тип("Число") Тогда

					ДанныеПечати.Вставить("Количество", Окр(СтрокаТабличнойЧасти.Количество
															* СтрокаТабличнойЧасти.КоэффициентЕдиницыИзмерения, 3));

				Иначе

					ДанныеПечати.Вставить("Количество", СтрокаТабличнойЧасти.Количество);

				КонецЕсли;

				Если ЕстьКолонкаАкциз И ЗначениеЗаполнено(СтрокаТабличнойЧасти.Акциз) Тогда
					ДанныеПечати.Вставить("Акциз", СтрокаТабличнойЧасти.Акциз);
				Иначе
					ДанныеПечати.Вставить("Акциз", ?(ИспользуетсяПостановление1137, НСтр("ru ='без акциза'"), ""));
				КонецЕсли;
				
				ДанныеПечати.Вставить("СтавкаНДС", СтрокаТабличнойЧасти.СтавкаНДС);

				СтруктураИтогов.ЕстьСтавкаНольПроцентов = (СтруктураИтогов.ЕстьСтавкаНольПроцентов
														   Или ДанныеПечати.СтавкаНДС = УправлениеНебольшойФирмойПовтИсп.ПолучитьСтавкуНДСНоль());

				Если ДанныеОбъекта.СуммаВключаетНДС Тогда

					СуммаБезНДС = СтрокаТабличнойЧасти.Сумма - СтрокаТабличнойЧасти.СуммаНДС;

				Иначе

					СуммаБезНДС = СтрокаТабличнойЧасти.Сумма;

				КонецЕсли;

				Если ТипЗнч(ДанныеПечати.Количество) <> Тип("Число") Тогда

					Цена = Неопределено; //При реализации услуг количество (и цена) могут не указываться

				ИначеЕсли ДанныеОбъекта.СуммаВключаетНДС = Ложь
						  И СтрокаТабличнойЧасти.КоэффициентЕдиницыИзмерения = 1
						  И СтрокаТабличнойЧасти.Владелец().Колонки.Найти("СуммаСкидкиПоСтроке") <> Неопределено
						  И СтрокаТабличнойЧасти.СуммаСкидкиПоСтроке = 0 Тогда

					Цена = СтрокаТабличнойЧасти.Цена;

				Иначе

					Цена = СуммаБезНДС / ?(ДанныеПечати.Количество = 0, 1, ДанныеПечати.Количество);

				КонецЕсли;

				ДанныеПечати.Вставить("Цена", Цена);
				ДанныеПечати.Вставить("Стоимость", СуммаБезНДС);
				ДанныеПечати.Вставить("СуммаНДС", СтрокаТабличнойЧасти.СуммаНДС);
				ДанныеПечати.Вставить("Всего", СтрокаТабличнойЧасти.Всего);

				ДанныеПечати.Вставить("СтранаСсылка", СтрокаТабличнойЧасти.СтранаСсылка);
				ДанныеПечати.Вставить("СтранаПроисхожденияКод", СтрокаТабличнойЧасти.СтранаКод);
				ДанныеПечати.Вставить("ПредставлениеСтраны", СтрокаТабличнойЧасти.СтранаПредставление);
				ДанныеПечати.Вставить("ПредставлениеГТД", СтрокаТабличнойЧасти.ПредставлениеГТД);
				
				ЭтоАванс = (ДанныеОбъекта.ВидОперации = Перечисления.ВидыОперацийСчетФактура.Аванс);
				ДанныеПечати.Вставить("ЭтоАванс", ЭтоАванс);

				ОбработатьПустыеЗначенияДанныхПечатиСтроки(ДанныеПечати, ДанныеОбъекта);
				ОбластиМакета.ОбластьСтроки.Параметры.Заполнить(ДанныеПечати);
				// Меняем только печатный параметр после расчёта цены в кг.
				Если КартаКоличествЛотковУПД <> Неопределено
					И ТипЗнч(ДанныеПечати.Количество) = Тип("Число") Тогда
					КоличествоЛотковУПД = КартаКоличествЛотковУПД[ДанныеОбъекта.ТаблицаЗапасы.Индекс(СтрокаТабличнойЧасти)];
					Если КоличествоЛотковУПД <> Неопределено Тогда
						ОбластиМакета.ОбластьСтроки.Параметры.Количество = КоличествоЛотковУПД;
					КонецЕсли;
				КонецЕсли;

				// Прослеживаемость
				Если ВыводитьРНПТ И ЗначениеЗаполнено(СтрокаТабличнойЧасти.СведенияОПрослеживаемости) Тогда
					
					ТаблицаРНПТ = СтрокаТабличнойЧасти.СведенияОПрослеживаемости;
					ПерваяСтрока = Истина;
					ОбластьЗапись.ГраницаСнизу  = Новый Линия(ТипЛинииЯчейкиТабличногоДокумента.НетЛинии);
					ОбластьЗапись.ГраницаСверху = Новый Линия(ТипЛинииЯчейкиТабличногоДокумента.НетЛинии);
					
					ЕстьКолонкаКодТНВЭДЕдиницаИзмеренияКод = НЕ ТаблицаРНПТ.Колонки.Найти("КодТНВЭДЕдиницаИзмеренияКод") = Неопределено;
					ЕстьКолонкаКодТНВЭДЕдиницаИзмеренияНаименование = НЕ ТаблицаРНПТ.Колонки.Найти("КодТНВЭДЕдиницаИзмеренияНаименование") = Неопределено;
					
					ЕстьРеквизитыЕдиницыИзмеренияВТаблицеРНПТ = ЕстьКолонкаКодТНВЭДЕдиницаИзмеренияКод И ЕстьКолонкаКодТНВЭДЕдиницаИзмеренияНаименование;
					ДанныеПоСтранеПроисхожденияИзРНПТ = Ложь;
					ЕстьОбластьОбластьЗаписьБезСтраныПроисхождения = НЕ ОбластьМакета.Области.Найти("ОбластьЗаписьБезСтраныПроисхождения") = Неопределено;
					
					Для ИндексСтроки = 0 По ТаблицаРНПТ.Количество() - 1 Цикл
						
						Если Не ПерваяСтрока Тогда
							// Получаем область для последующих строк для очистки параметров.
							ОбластьМакета = Макет.ПолучитьОбласть("Строка");
						КонецЕсли;
						
						ОбластьМакета.Параметры.ПредставлениеГТД           = ТаблицаРНПТ[ИндексСтроки].РНПТ;
						
						Если ЕстьРеквизитыЕдиницыИзмеренияВТаблицеРНПТ

				ОбластьБезФаксимиле = ТабличныйДокумент.Области.Найти(ИмяСкрываемойОбласти);
				Если ОбластьБезФаксимиле <> Неопределено Тогда
					ОбластьБезФаксимиле.Имя = ОбластьБезФаксимиле.Имя + ИмяОбласти;
					Если Не ЕстьФаксимиле Тогда
						ТабличныйДокумент.УдалитьОбласть(ОбластьБезФаксимиле, ТипСмещенияТабличногоДокумента.ПоВертикали);
					КонецЕсли;
				КонецЕсли;
				
			КонецЕсли;
			
		КонецЕсли;
	
	КонецЦикла;
	
	Возврат ТабличныйДокумент;

КонецФункции

// Печать счета-фактуры.
// 
// Параметры:
//  МассивОбъектов - Массив
//  ПараметрыПечати - Структура
//  КоллекцияПечатныхФорм - ТаблицаЗначений
//  ОбъектыПечати - СписокЗначений
//  ПараметрыВывода - Структура
Процедура Печать(МассивОбъектов, ПараметрыПечати, КоллекцияПечатныхФорм, ОбъектыПечати, ПараметрыВывода) Экспорт
	Перем Ошибки;

	ПечатнаяФорма = УправлениеПечатью.СведенияОПечатнойФорме(КоллекцияПечатныхФорм, "СчетФактура");
	Если ПечатнаяФорма <> Неопределено Тогда

		ПечатнаяФорма.ТабличныйДокумент = Новый ТабличныйДокумент;
		ПечатнаяФорма.ТабличныйДокумент.КлючПараметровПечати = Обработки.ПечатьСчетФактура.КлючПараметровПечати(Ложь);
		ПечатнаяФорма.ПолныйПутьКМакету = Неопределено;
		ПечатнаяФорма.СинонимМакета = Обработки.ПечатьСчетФактура.СинонимМакета(Ложь);

		ТаблицаОписанияСчетовФактур = РазобратьДокументыНаСчетаФактуры(МассивОбъектов, Ложь);
		ЗаполнитьДанныеДокументов(ТаблицаОписанияСчетовФактур, ОбъектыПечати, Ложь, Ложь, Ошибки);

		ВывестиТабличныеДокументы(ПечатнаяФорма, ТаблицаОписанияСчетовФактур, ОбъектыПечати, Ложь);

	КонецЕсли;

	ПечатнаяФорма = УправлениеПечатью.СведенияОПечатнойФорме(КоллекцияПечатныхФорм, "УниверсальныйПередаточныйДокумент");
	Если ПечатнаяФорма <> Неопределено Тогда

		ПечатнаяФорма.ТабличныйДокумент = Новый ТабличныйДокумент;
		ПечатнаяФорма.ТабличныйДокумент.КлючПараметровПечати = Обработки.ПечатьСчетФактура.КлючПараметровПечати(Истина);
		ПечатнаяФорма.ПолныйПутьКМакету = Неопределено;
		ПечатнаяФорма.СинонимМакета = Обработки.ПечатьСчетФактура.СинонимМакета(Истина);

		ТаблицаОписанияСчетовФактур = РазобратьДокументыНаСчетаФактуры(МассивОбъектов, Истина);
		ЗаполнитьДанныеДокументов(ТаблицаОписанияСчетовФактур, ОбъектыПечати, Истина, Ложь, Ошибки);

		ВывестиТабличныеДокументы(ПечатнаяФорма, ТаблицаОписанияСчетовФактур, ОбъектыПечати, Истина);

	КонецЕсли;
	
	ПечатнаяФорма = УправлениеПечатью.СведенияОПечатнойФорме(КоллекцияПечатныхФорм, "СчетФактураПолученный");
	Если ПечатнаяФорма <> Неопределено Тогда

		ПечатнаяФорма.ТабличныйДокумент = Новый ТабличныйДокумент;
		ПечатнаяФорма.ТабличныйДокумент.КлючПараметровПечати = Обработки.ПечатьСчетФактура.КлючПараметровПечати(Ложь);
		ПечатнаяФорма.ПолныйПутьКМакету = Неопределено;
		ПечатнаяФорма.СинонимМакета = Обработки.ПечатьСчетФактура.СинонимМакета(Ложь);

		ТаблицаОписанияСчетовФактур = РазобратьДокументыНаСчетаФактуры(МассивОбъектов, Ложь, Истина);
		ЗаполнитьДанныеДокументов(ТаблицаОписанияСчетовФактур, ОбъектыПечати, Ложь, Ложь, Ошибки);

		ВывестиТабличныеДокументы(ПечатнаяФорма, ТаблицаОписанияСчетовФактур, ОбъектыПечати, Ложь);

	КонецЕсли;
	
	ПечатнаяФорма = УправлениеПечатью.СведенияОПечатнойФорме(КоллекцияПечатныхФорм, "СчетФактураВозвратОтПокупателя");
	Если ПечатнаяФорма <> Неопределено Тогда

		ПечатнаяФорма.ТабличныйДокумент = Новый ТабличныйДокумент;
		ПечатнаяФорма.ТабличныйДокумент.КлючПараметровПечати = Обработки.ПечатьСчетФактура.КлючПараметровПечати(Ложь);
		ПечатнаяФорма.ПолныйПутьКМакету = Неопределено;
		ПечатнаяФорма.СинонимМакета = Обработки.ПечатьСчетФактура.СинонимМакета(Ложь);

		ТаблицаОписанияСчетовФактур = РазобратьДокументыНаСчетаФактуры(МассивОбъектов, Ложь, Ложь);
		ЗаполнитьДанныеДокументов(ТаблицаОписанияСчетовФактур, ОбъектыПечати, Ложь, Ложь, Ошибки);

		ВывестиТабличныеДокументы(ПечатнаяФорма, ТаблицаОписанияСчетовФактур, ОбъектыПечати, Ложь);

	КонецЕсли;

	Если Ошибки <> Неопределено Тогда

		ОбщегоНазначенияКлиентСервер.СообщитьОшибкиПользователю(Ошибки);

	КонецЕсли;
	
	// параметры отправки печатных форм по электронной почте
	ЭлектроннаяПочтаУНФ.ЗаполнитьПараметрыОтправки(ПараметрыВывода.ПараметрыОтправки, МассивОбъектов,
		КоллекцияПечатныхФорм);

КонецПроцедуры

Функция ЕстьЗапасы(ДанныеОбъекта)

	Если ТипЗнч(ДанныеОбъекта) = Тип("СтрокаТаблицыЗначений")
		И ДанныеОбъекта.Владелец().Колонки.Найти("ЕстьЗапасы") <> Неопределено Тогда
		УказанГрузоотправительИлиГрузополучатель = ЗначениеЗаполнено(ДанныеОбъекта.Грузоотправитель) Или ЗначениеЗаполнено(ДанныеОбъекта.Грузополучатель);
		
		Возврат ДанныеОбъекта.ЕстьЗапасы Или УказанГрузоотправительИлиГрузополучатель;
	Иначе
		ТабличнаяЧастьДокумента = ДанныеОбъекта.ТаблицаЗапасы;
		
		Возврат (ТабличнаяЧастьДокумента.Количество() > 0) И (ТабличнаяЧастьДокумента.Найти(
			Перечисления.ТипыНоменклатуры.Запас) <> Неопределено);
	КонецЕсли;

Процедура УчестьРазличияСводногоСФ(ДанныеОбъекта, ДанныеПечати, ЭтоУПД)

	Если Не ДанныеОбъекта.ЭтоСводныйСчетФактура Тогда

		Возврат;

	КонецЕсли;

	ПредставлениеПокупателя = "";
	ЗначениеАдресаПокупателя = "";
	ПредставлениеГрузополучателя = "";
	ИНН_КПП_Покупателя = "";

	Для Каждого Строка Из ДанныеОбъекта.ТаблицаПокупателей Цикл

		ПредставлениеПокупателя = ПредставлениеПокупателя + ?(ПустаяСтрока(ПредставлениеПокупателя), "", "; ")
			+ ПечатьДокументовУНФ.ОписаниеОрганизации(Строка.СведенияОПокупателе, "ПолноеНаименование,");

		КПП = ПечатьДокументовУНФ.ОписаниеОрганизации(Строка.СведенияОПокупателе, "КПП,", Ложь);
		ИНН_КПП_Покупателя = ИНН_КПП_Покупателя + ?(ПустаяСтрока(ИНН_КПП_Покупателя), "", "; ")
						+ ПечатьДокументовУНФ.ОписаниеОрганизации(Строка.СведенияОПокупателе, "ИНН,", Ложь)
						+ ?(ПустаяСтрока(КПП), "", "/" + КПП);

		ЗначениеАдресаПокупателя = ЗначениеАдресаПокупателя + ?(ПустаяСтрока(ЗначениеАдресаПокупателя), "", "; ")
			+ ПечатьДокументовУНФ.ОписаниеОрганизации(Строка.СведенияОПокупателе, "ЮридическийАдрес,");

		ПредставлениеГрузополучателя = ПредставлениеГрузополучателя + ?(ПустаяСтрока(ПредставлениеГрузополучателя), "",
			"; ") + ПечатьДокументовУНФ.ОписаниеОрганизации(Строка.СведенияОГрузополучателе,
			"ПолноеНаименование, ФактическийАдрес,");

	КонецЦикла;

	ДанныеПечати.Вставить("ПредставлениеПокупателя", ПредставлениеПокупателя);
	ДанныеПечати.Вставить("АдресПокупателя", ЗначениеАдресаПокупателя);
	ДанныеПечати.Вставить("ПредставлениеГрузополучателя", ПредставлениеГрузополучателя);
	ДанныеПечати.Вставить("ИННПокупателя", ИНН_КПП_Покупателя);

КонецПроцедуры

Процедура ОбработатьПустыеПоляШапки(ДанныеПечати, ЕстьЗапасы, ЭтоУПД)

	Прочерк = НСтр("ru ='--'");
	ЭтоПродажаЗапасов = ЕстьЗапасы И (ДанныеПечати.ВидОперации = Перечисления.ВидыОперацийСчетФактура.Продажа);

	Если ПустаяСтрока(ДанныеПечати.ПредставлениеГрузоотправителя) Тогда

		ДанныеПечати.ПредставлениеГрузоотправителя = ?(ЭтоПродажаЗапасов, НСтр("ru ='Он же'"), Прочерк);

	КонецЕсли;

	Если ПустаяСтрока(ДанныеПечати.ПредставлениеГрузополучателя) Тогда

		ДанныеПечати.ПредставлениеГрузополучателя = ?(ЭтоПродажаЗапасов, ДанныеПечати.ПредставлениеПокупателя, Прочерк);

	КонецЕсли;

	Если Не ЗначениеЗаполнено(ДанныеПечати.НомерИсправления) Тогда

		ДанныеПечати.НомерИсправления = Прочерк;
		ДанныеПечати.ДатаИсправления = Прочерк;

	КонецЕсли;

	Если ПустаяСтрока(ДанныеПечати.ПоДокументу) Тогда

		МассивВидовОпераций = Новый Массив;
		МассивВидовОпераций.Добавить(Перечисления.ВидыОперацийСчетФактура.НаАвансКомитента);
		МассивВидовОпераций.Добавить(Перечисления.ВидыОперацийСчетФактура.Аванс);

		ШаблонТекста = НСтр("ru ='№ %1 от %1'");
		ДанныеПечати.ПоДокументу = ДанныеПечати.ПоДокументу + Символы.НПП + СтрШаблон(ШаблонТекста, ?(
			МассивВидовОпераций.Найти(ДанныеПечати.ВидОперации) <> Неопределено, Прочерк, Символы.НПП));

	КонецЕсли;

	Если ДанныеПечати.ВидОперации <> Перечисления.ВидыОперацийСчетФактура.Аванс Тогда

		Если ПустаяСтрока(ДанныеПечати.АдресПокупателя) Тогда

			ДанныеПечати.АдресПокупателя = ДанныеПечати.АдресПокупателя + Прочерк;

		КонецЕсли;

		Если ПустаяСтрока(ДанныеПечати.ПредставлениеГрузоотправителя) Тогда

			ДанныеПечати.ПредставлениеГрузоотправителя = ДанныеПечати.ПредставлениеГрузоотправителя + Прочерк;

		КонецЕсли;

		Если ПустаяСтрока(ДанныеПечати.ПредставлениеГрузополучателя) Тогда

			ДанныеПечати.ПредставлениеГрузополучателя = ДанныеПечати.ПредставлениеГрузополучателя + Прочерк;

		КонецЕсли;

	КонецЕсли;

	Если ПустаяСтрока(ДанныеПечати.Валюта) Тогда

		ДанныеПечати.Валюта = НСтр("ru ='Российский рубль, 643'");

	КонецЕсли;

КонецПроцедуры

Процедура ОбработатьПустыеЗначенияДанныхПечатиСтроки(ДанныеПечати, ДанныеОбъекта)

	СтандартныеИменаПолей = "ЕдиницаИзмерения,ЕдиницаИзмеренияКод,Количество,Цена,СуммаНДС,СтранаПроисхожденияКод,ПредставлениеСтраны,ПредставлениеГТД";
	
	// Для наборов выводим только расшифровку, числовые значения не нужны.
	//
	Если ДанныеПечати.ЭтоНабор = Истина Тогда

		ДанныеПечати.Вставить("Акциз", Неопределено);
		ДанныеПечати.Вставить("Цена", Неопределено);
		ДанныеПечати.Вставить("Стоимость", Неопределено);
		ДанныеПечати.Вставить("СуммаНДС", Неопределено);
		ДанныеПечати.Вставить("Всего", Неопределено);

		СтандартныеИменаПолей = СтрЗаменить(СтандартныеИменаПолей, "Цена,", "");

	КонецЕсли;

	Если ДанныеПечати.ЭтоАванс = Истина Тогда

		ДанныеПечати.Вставить("Акциз", Неопределено);
		ДанныеПечати.Вставить("ЕдиницаИзмерения", Неопределено);
		ДанныеПечати.Вставить("ЕдиницаИзмеренияКод", Неопределено);
		ДанныеПечати.Вставить("Количество", Неопределено);
		ДанныеПечати.Вставить("ПредставлениеКодаНоменклатуры", Неопределено);
		ДанныеПечати.Вставить("Цена", Неопределено);
		ДанныеПечати.Вставить("Стоимость", Неопределено);
		ДанныеПечати.Вставить("СтранаПроисхожденияКод", Неопределено);
		ДанныеПечати.Вставить("СтранаСсылка", Неопределено);

		СтандартныеИменаПолей = СтандартныеИменаПолей + ",Стоимость,Акциз";

	КонецЕсли;
	
	// Сумма налога, предъявляемая покупателю при ставке ндс 0% должна заполняться как 0,00 (п.5 ст.169 НК РФ)
	//
	Если ДанныеПечати.Свойство("СтавкаНДС")
		 И ДанныеПечати.СтавкаНДС = УправлениеНебольшойФирмойПовтИсп.ПолучитьСтавкуНДСНоль() Тогда

		СтандартныеИменаПолей = СтрЗаменить(СтандартныеИменаПолей, "СуммаНДС,", "");

	КонецЕсли;
	
	// Родная страна происхождения (РОССИЯ) не требует указания в документах (СФ, УПД).
	//
	Если ДанныеПечати.СтранаСсылка = Справочники.СтраныМира.Россия Или ПустаяСтрока(ДанныеПечати.ПредставлениеСтраны) Тогда

		ДанныеПечати.Вставить("СтранаПроисхожденияКод", Неопределено);
		ДанныеПечати.Вставить("ПредставлениеСтраны", Неопределено);

	КонецЕсли;

	Если ДанныеПечати.Свойство("ТоварКодТНВЭД") И Не ЗначениеЗаполнено(ДанныеПечати.ТоварКодТНВЭД) Тогда

		ДанныеПечати.ТоварКодТНВЭД = Неопределено;
		СтандартныеИменаПолей = СтандартныеИменаПолей + ",ТоварКодТНВЭД";

	КонецЕсли;

	ЗаменитьПустыеЗначенияНаПрочерки(ДанныеПечати, СтандартныеИменаПолей);
	
	// Ставка "Без НДС"
	Если ДанныеПечати.Свойство("СтавкаНДС")
		И ДанныеПечати.СтавкаНДС = УправлениеНебольшойФирмойПовтИсп.ПолучитьСтавкуНДСБезНДС() Тогда
		
		Если ДанныеОбъекта.Владелец().Колонки.Найти("ЭтоУниверсальныйДокумент") <> Неопределено
			И ДанныеОбъекта.Владелец().Колонки.Найти("НалогообложениеНДС") <> Неопределено Тогда
			НужноУказыватьСтавкуБезНДС = (ДанныеОбъекта.НалогообложениеНДС = ПредопределенноеЗначение("Перечисление.ТипыНалогообложенияНДС.ОблагаетсяНДС"));
			Если НужноУказыватьСтавкуБезНДС Тогда
				ДанныеПечати.Вставить("СуммаНДС", НСтр("ru='Без НДС'"));
			КонецЕсли;
		КонецЕсли;
		
	КонецЕсли;

КонецПроцедуры

Процедура ЗаменитьПустыеЗначенияНаПрочерки(ДанныеПечати, ИменаПолей)

	Если Не ТипЗнч(ДанныеПечати) = Тип("Структура") Тогда

		Возврат;

	КонецЕсли;

	МассивИмен = Неопределено;
	Если Не ПустаяСтрока(ИменаПолей) Тогда

		МассивИмен = СтрРазделить(ИменаПолей, ",");

	КонецЕсли;

	Для Каждого ЭлементСтруктуры Из ДанныеПечати Цикл

		Если ЗначениеЗаполнено(ЭлементСтруктуры.Значение) Тогда

			Продолжить;

		КонецЕсли;

		Если МассивИмен <> Неопределено И МассивИмен.Найти(ЭлементСтруктуры.Ключ) = Неопределено Тогда

			Продолжить;

		КонецЕсли;

		ДанныеПечати[ЭлементСтруктуры.Ключ] = "--";

	КонецЦикла;

КонецПроцедуры

Процедура УвеличитьИтогиДокумента(СтруктураИтогов, ДанныеПечати)

	Если ДанныеПечати.ЭтоНабор = Истина Тогда

		Возврат;

	КонецЕсли;

	Если ТипЗнч(ДанныеПечати.Стоимость) = Тип("Число") Тогда

		СтруктураИтогов.ПоСтраницеСтоимость = СтруктураИтогов.ПоСтраницеСтоимость + ДанныеПечати.Стоимость;
		СтруктураИтогов.Стоимость = СтруктураИтогов.Стоимость + ДанныеПечати.Стоимость;

	КонецЕсли;

	Если ТипЗнч(ДанныеПечати.СуммаНДС) = Тип("Число") Тогда

		СтруктураИтогов.ПоСтраницеСуммаНДС = СтруктураИтогов.ПоСтраницеСуммаНДС + ДанныеПечати.СуммаНДС;
		СтруктураИтогов.СуммаНДС = СтруктураИтогов.СуммаНДС + ДанныеПечати.СуммаНДС;

	КонецЕсли;

	Если ТипЗнч(ДанныеПечати.Всего) = Тип("Число") Тогда

		СтруктураИтогов.ПоСтраницеВсего = СтруктураИтогов.ПоСтраницеВсего + ДанныеПечати.Всего;
		СтруктураИтогов.Всего = СтруктураИтогов.Всего + ДанныеПечати.Всего;

	КонецЕсли;

КонецПроцедуры

Функция СформироватьПредставлениеОснования(ДанныеПечати)

	Основание = "";
	Если Не ПустаяСтрока(ДанныеПечати.ОснованиеПечати) Тогда

		Основание = ДанныеПечати.ОснованиеПечати;

	КонецЕсли;

	ДанныеОДоверенности = СокрЛП(ДанныеПечати.ДоверенностьНомер) + СокрЛП(ДанныеПечати.ДоверенностьВыдана) + СокрЛП(
		ДанныеПечати.ДоверенностьЛицо);
	Если Не ПустаяСтрока(ДанныеОДоверенности) Или ЗначениеЗаполнено(ДанныеПечати.ДоверенностьДата) Тогда

		Основание = Основание
					+ ?(ПустаяСтрока(Основание), "", ";")
					+ НСтр("ru =' по доверенности №'")
					+ ДанныеПечати.ДоверенностьНомер
					+ НСтр("ru =' от '")
					+ Формат(ДанныеПечати.ДоверенностьДата, "ДЛФ=DD")
					+ НСтр("ru =' выданной '")
					+ ДанныеПечати.ДоверенностьВыдана
					+ Символы.НПП
					+ ДанныеПечати.ДоверенностьЛицо;

	КонецЕсли;

	Возврат Основание;

		Если ПустаяСтрока(СведенияОСубъекте.КПП) Тогда

			ПредставлениеСубъекта = СтроковыеФункцииКлиентСервер.ПодставитьПараметрыВСтроку(НСтр("ru = '%1, ИНН %2'"),

			ПредставлениеСубъекта = СтроковыеФункцииКлиентСервер.ПодставитьПараметрыВСтроку(НСтр(
				"ru = '%1, ИНН/КПП %2/%3'"), ПредставлениеСубъекта, СведенияОСубъекте.ИНН, СведенияОСубъекте.КПП);

Процедура УчестьОсобенностиДокументаСчетФактура(ТаблицаДанных, СчетФактура, ДополнительныеСвойства)
	
	ОсобенностиЗаполнения = Новый Структура("ЭтоЗаказНаряд, ЭтоОтчетКомиссионера, ЭтоОтчетОПереработке, ЭтоПодтверждениеНулевойСтавкиНДС", Ложь, Ложь, Ложь, Ложь);
	ЗаполнитьЗначенияСвойств(ОсобенностиЗаполнения, ДополнительныеСвойства);
	
	ТаблицаДанных.Колонки.Добавить("ПлатежныеДокументы");
	ТаблицаДанных.Колонки.Добавить("ИдентификаторГосКонтракта", Новый ОписаниеТипов("Строка"));

	ДанныеСчетаФактуры = РеквизитыСчетаФактурыДляПечати(СчетФактура);

	ТаблицаДанных[0].ПлатежныеДокументы = ДанныеСчетаФактуры.ПлатежныеДокументы.Скопировать();
	ТаблицаДанных[0].ИдентификаторГосКонтракта = ДанныеСчетаФактуры.ИдентификаторГосКонтракта;
	
	ЕстьПолеОригинальныйСчетФактура = ТаблицаДанных.Колонки.Найти("ОригинальныйСчетФактура") <> Неопределено;

	// В счет-фактуре может быть указан номер для печати, его выводим без изменений
	Если ТаблицаДанных.Колонки.Найти("НомерДляПечати") = Неопределено Тогда
		ТаблицаДанных.Колонки.Добавить("НомерДляПечати", Новый ОписаниеТипов("Строка", , Новый КвалификаторыСтроки(50)));
	КонецЕсли;
	
	Если ОсобенностиЗаполнения.ЭтоОтчетОПереработке Тогда

		ТаблицаДанных.Колонки.ТаблицаПродукция.Имя = "ТаблицаЗапасы";

	КонецЕсли;
	
	Для Каждого СтрокаТаблицы Из ТаблицаДанных Цикл

		Если СтрокаТаблицы.ЭтоКорректировка Тогда

			Если ЕстьПолеОригинальныйСчетФактура Тогда
				
				Данные = СчетаФактурыУНФ.ДатаНомерДляПечатиСчетаФактуры(СтрокаТаблицы.ОригинальныйСчетФактура);
				СтрокаТаблицы.НомерДляПечати = Данные.Номер;
				СтрокаТаблицы.ДатаДокумента = Данные.Дата;

			КонецЕсли;

		ИначеЕсли ЗначениеЗаполнено(СчетФактура) Тогда
			
			Данные = СчетаФактурыУНФ.ДатаНомерДляПечатиСчетаФактуры(СчетФактура);
			
			СтрокаТаблицы.НомерДляПечати = Данные.Номер;
			СтрокаТаблицы.ДатаДокумента = Данные.Дата;
			
			Если ЗначениеЗаполнено(ДанныеСчетаФактуры.Грузоотправитель) Тогда
				СтрокаТаблицы.Грузоотправитель = ДанныеСчетаФактуры.Грузоотправитель;
			КонецЕсли;

			Если ЗначениеЗаполнено(ДанныеСчетаФактуры.Грузополучатель) Тогда
				СтрокаТаблицы.Грузополучатель = ДанныеСчетаФактуры.Грузополучатель;
			КонецЕсли;
			
		Иначе
			
			СтрокаТаблицы.НомерДляПечати = ПечатьДокументовУНФ.ПолучитьНомерНаПечатьСУчетомДатыДокумента(
				СтрокаТаблицы.ДатаДокумента, СтрокаТаблицы.Номер, СтрокаТаблицы.Префикс)
			
		КонецЕсли;

		ЗаполнитьРеквизитПоСчетуФактуре(СтрокаТаблицы, "ДолжностьРуководителя", ДанныеСчетаФактуры.ДолжностьРуководителя);
		ЗаполнитьРеквизитПоСчетуФактуре(СтрокаТаблицы, "ДолжностьКладовщика", ДанныеСчетаФактуры.ДолжностьКладовщика);
		ЗаполнитьРеквизитПоСчетуФактуре(СтрокаТаблицы, "РасшифровкаПодписиРуководителя", ДанныеСчетаФактуры.РасшифровкаПодписиРуководителя);
		ЗаполнитьРеквизитПоСчетуФактуре(СтрокаТаблицы, "РасшифровкаПодписиГлавногоБухгалтера", ДанныеСчетаФактуры.РасшифровкаПодписиГлавногоБухгалтера); 
		ЗаполнитьРеквизитПоСчетуФактуре(СтрокаТаблицы, "РасшифровкаПодписиКладовщика", ДанныеСчетаФактуры.РасшифровкаПодписиКладовщика);
		ЗаполнитьРеквизитПоСчетуФактуре(СтрокаТаблицы, "ФаксимилеРуководителя", ДанныеСчетаФактуры.ФаксимилеРуководителя);
		ЗаполнитьРеквизитПоСчетуФактуре(СтрокаТаблицы, "ФаксимилеГлавногоБухгалтера", ДанныеСчетаФактуры.ФаксимилеГлавногоБухгалтера);
		ЗаполнитьРеквизитПоСчетуФактуре(СтрокаТаблицы, "ФаксимилеКладовщика", ДанныеСчетаФактуры.ФаксимилеКладовщика);
		
		Если ДанныеСчетаФактуры.СводныйКомиссионный Тогда
			
			СтрокаТаблицы.ЭтоСводныйСчетФактура = Истина;

			Для Каждого СтрокаПокупателя Из СтрокаТаблицы.ТаблицаПокупателей Цикл

				СтрокаПокупателя.СведенияОПокупателе = ПечатьДокументовУНФ.СведенияОЮрФизЛице(
					СтрокаПокупателя.Покупатель, СтрокаТаблицы.ДатаДокумента, , );
				СтрокаПокупателя.СведенияОГрузополучателе = ПечатьДокументовУНФ.СведенияОЮрФизЛице(
					СтрокаПокупателя.Покупатель, СтрокаТаблицы.ДатаДокумента, , );

			КонецЦикла;
			
		КонецЕсли;

		Если ОсобенностиЗаполнения.ЭтоЗаказНаряд Тогда
			
			Для Каждого СтрокаРаботыУслуги Из СтрокаТаблицы.ТаблицаРаботыУслуги Цикл
				ЗаполнитьЗначенияСвойств(СтрокаТаблицы.ТаблицаЗапасы.Добавить(), СтрокаРаботыУслуги);
			КонецЦикла;
			
		КонецЕсли;
		
		Если ДанныеСчетаФактуры.ДокументыОснования <> Неопределено Тогда
			
			СтрокаОснования = ДанныеСчетаФактуры.ДокументыОснования.Найти(СтрокаТаблицы.Ссылка, "ДокументОснование");
			Если СтрокаОснования <> Неопределено И ЗначениеЗаполнено(СтрокаОснования.НаименованиеДокументаОтгрузки) Тогда
				СтрокаТаблицы.ТаблицаЗапасы.ЗаполнитьЗначения(СтрокаОснования.НаименованиеДокументаОтгрузки, "НаименованиеОтгрузочногоДокумента");
			КонецЕсли;
			
		КонецЕсли;
		
	КонецЦикла;

	Если ОсобенностиЗаполнения.ЭтоОтчетКомиссионера Тогда

		ПараметрыОтбора = Новый Структура("СчетФактура",СчетФактура);
		ПараметрыОтбораКлючСвязи = Новый Структура("КлючСвязи");
		
		Для Каждого СтрокаТаблицы Из ТаблицаДанных Цикл

			ТаблицаЗапасов = СтрокаТаблицы.ТаблицаЗапасы.СкопироватьКолонки();
			НайденныеСтроки = СтрокаТаблицы.ТаблицаПокупателей.НайтиСтроки(ПараметрыОтбора);
			
			Если Не НайденныеСтроки.Количество() Тогда
				Продолжить;
			КонецЕсли;
			
			СтрокаТаблицы.Грузополучатель = НайденныеСтроки[0].Покупатель;	
			СтрокаТаблицы.Контрагент = НайденныеСтроки[0].Покупатель;
			
			Если Не СтрокаТаблицы.ВыписыватьСчетаФактурыСводно Тогда
				
				СтрокаПокупателяССчетФактурой = НайденныеСтроки[0];
				ПараметрыОтбораКлючСвязи.КлючСвязи = СтрокаПокупателяССчетФактурой.КлючСвязи;
				НайденныеСтрокиЗапасов = СтрокаТаблицы.ТаблицаЗапасы.НайтиСтроки(ПараметрыОтбораКлючСвязи);
				
				Для Каждого НайденнаяСтрокаЗапасы Из НайденныеСтрокиЗапасов Цикл
					НоваяСтрока = ТаблицаЗапасов.Добавить();
					ЗаполнитьЗначенияСвойств(НоваяСтрока, НайденнаяСтрокаЗапасы);
				КонецЦикла;
				
			Иначе
				
				Для Каждого НайденнаяСтрока Из НайденныеСтроки Цикл
					
					ПараметрыОтбораКлючСвязи.КлючСвязи = НайденнаяСтрока.КлючСвязи;
					НайденныеСтрокиЗапасов = СтрокаТаблицы.ТаблицаЗапасы.НайтиСтроки(ПараметрыОтбораКлючСвязи);

	ТаблицаОписанияСчетовФактур.Колонки.Добавить("ЭтоВходящийДокумент", Неопределено); // Для УПД без счетов-фактур
	ТаблицаОписанияСчетовФактур.Колонки.Добавить("ОснованиеУПД", Неопределено); // Для УПД без счетов-фактур

	ТаблицаОписанияСчетовФактур.Колонки.Добавить("Постановление", Неопределено);

		Если ТипЗнч(ОбъектПечати) = Тип("ДокументСсылка.СчетФактура") Или ТипЗнч(ОбъектПечати) = Тип(
			"ДокументСсылка.СчетФактураПолученный") Тогда

			СтруктураДанных = Новый Структура;
			СтруктураДанных.Вставить("Ссылка", ОбъектПечати);
			СтруктураДанных.Вставить("Версия", ?(СтруктураДанных.Ссылка.УдалитьЗапасы.Количество() > 0, "1.6.11",
				"1.6.12"));

		Иначе

			СтруктураДанных = СчетаФактурыУНФ.ПолучитьПодчиненныйСчетФактуру(ОбъектПечати,
				ЭтоВходящийДокумент);

		КонецЕсли;

		ОписаниеСчетФактуры.ЭтоВходящийДокумент = ЭтоВходящийДокумент;
		ОписаниеСчетФактуры.ПФТолькоВРублях = Истина;

		Если СтруктураДанных <> Неопределено Тогда

			ОписаниеСчетФактуры.СчетФактура = СтруктураДанных.Ссылка;
			ОписаниеСчетФактуры.ВерсияДокумента = СтруктураДанных.Версия;

			Если ЗначениеЗаполнено(ОписаниеСчетФактуры.СчетФактура.Договор)
				 И ОписаниеСчетФактуры.СчетФактура.Договор.ВалютаРасчетов <> Константы.НациональнаяВалюта.Получить() Тогда

				ОписаниеСчетФактуры.ПФТолькоВРублях = ОписаниеСчетФактуры.СчетФактура.Договор.РасчетыВУсловныхЕдиницах;

			КонецЕсли;

			Если ЗначениеЗаполнено(ОписаниеСчетФактуры.СчетФактура) Тогда

				Если ОписаниеСчетФактуры.СчетФактура.ВидОперации = Перечисления.ВидыОперацийСчетФактура.Аванс
					Или ОписаниеСчетФактуры.СчетФактура.ВидОперации = Перечисления.ВидыОперацийСчетФактура.НаАвансКомитента
					Или ОписаниеСчетФактуры.СчетФактура.ВидОперации = Перечисления.ВидыОперацийСчетФактураПолученный.Аванс Тогда

					ОписаниеСчетФактуры.ОписаниеДокумента = "ЭтоАванс";

				ИначеЕсли ОписаниеСчетФактуры.СчетФактура.ВидОперации = Перечисления.ВидыОперацийСчетФактура.СуммовыеРазницы Тогда

					ОписаниеСчетФактуры.ОписаниеДокумента = "ЭтоСуммоваяРазница";

					ОписаниеСчетФактуры.ОписаниеДокумента = "НетВозможностиКорректноСформироватьДокумент";

				ИначеЕсли ОписаниеСчетФактуры.СчетФактура.ВидОперации = Перечисления.ВидыОперацийСчетФактура.Корректировка
						  Или ОписаниеСчетФактуры.СчетФактура.ВидОперации = Перечисления.ВидыОперацийСчетФактураПолученный.Корректировка Тогда

					ОписаниеСчетФактуры.ОписаниеДокумента = "ЭтоКорректировка";

				ИначеЕсли ЭтоВходящийДокумент Тогда

					Для Каждого ОписаниеОснование Из ОписаниеСчетФактуры.СчетФактура.ДокументыОснования Цикл

						Если ТипЗнч(ОписаниеОснование.ДокументОснование) = Тип("ДокументСсылка.ОтчетКомитенту") Тогда

							ОписаниеСчетФактуры.ОписаниеДокумента = "НетВозможностиКорректноСформироватьДокумент";
							Прервать;

							И ОписаниеОснование.ДокументОснование.ИспользоватьНовыйВидДокумента Тогда

							НайденныеСтрокиПокупателиВозвраты = ОписаниеОснование.ДокументОснование.ПокупателиВозвраты.НайтиСтроки(ПараметрыОтбора);
							
							Если НайденныеСтрокиПокупателиВозвраты.Количество() Тогда
								ОписаниеСчетФактуры.ОписаниеДокумента = "НетВозможностиКорректноСформироватьДокумент";
								Прервать;
							КонецЕсли;
							
						ИначеЕсли ТипЗнч(ОписаниеОснование.ДокументОснование) = Тип("ДокументСсылка.ОтчетКомиссионера")

Функция ТаблицаДанныхВходящегоДокументаФормата1_6_12(СтрокаОписаниеСФ, ИспользоватьФаксимиле, ЭтоУПД, Ошибки)
	ДанныеДокумента = Неопределено;

	Для Каждого СтрокаОснования Из СтрокаОписаниеСФ.СчетФактура.ДокументыОснования Цикл

		МассивДокументов = Новый Массив;
		МассивДокументов.Добавить(СтрокаОснования.ДокументОснование);

		Если СтрокаОписаниеСФ.ОписаниеДокумента = "ЭтоАванс" Тогда

			Если Не ЭтоУПД Тогда

				МенеджерДокумента = ОбщегоНазначения.МенеджерОбъектаПоСсылке(СтрокаОписаниеСФ.СчетФактура);
				ТаблицаДанных = МенеджерДокумента.ДанныеПолученныхАвансовыхДокументовРегУчет(
					СтрокаОписаниеСФ.СчетФактура, ИспользоватьФаксимиле);
				Прервать;

			КонецЕсли;

		ИначеЕсли СтрокаОписаниеСФ.ОписаниеДокумента = "ЭтоКорректировка" Тогда

			Если ЭтоУПД Тогда

				ТаблицаДанных = Неопределено;

			Иначе

				ЭтоУниверсальныйКорректировочныйДокумент = Ложь;
				ТекстЗапроса = Документы.СчетФактураПолученный.ТекстЗапросаПечатьКорректировочныхСчетовФактур();
				ТаблицаДанных = Документы.СчетФактура.ПолучитьДанныеДляПечатиКорректировочныхСчетовФактур(
					СтрокаОписаниеСФ.СчетФактура, ТекстЗапроса, ЭтоУниверсальныйКорректировочныйДокумент);

				Если ТаблицаДанных.Количество() < 1 Тогда

					ТаблицаДанных = Неопределено;

				КонецЕсли;

			КонецЕсли;

		ИначеЕсли СтрокаОписаниеСФ.ОписаниеДокумента = "СчетФактураВТабличнойЧасти"
			И ТипЗнч(СтрокаОснования.ДокументОснование) = Тип("ДокументСсылка.АвансовыйОтчет") Тогда
			
			СтруктураПараметров = Новый Структура;
			СтруктураПараметров.Вставить("МассивОбъектов", МассивДокументов);
			СтруктураПараметров.Вставить("СчетФактура", СтрокаОписаниеСФ.СчетФактура);
			СтруктураПараметров.Вставить("ИспользоватьФаксимиле", Ложь);
			СтруктураПараметров.Вставить("ПечатнаяФормаТолькоВРублях", СтрокаОписаниеСФ.ПФТолькоВРублях);
			СтруктураПараметров.Вставить("Ошибки", Ошибки);
			
			ТаблицаДанных = Документы.АвансовыйОтчет.ДанныеПолученныхДокументовРегУчетСчетФактураВТабличнойЧасти(СтруктураПараметров);
			
			УчестьОсобенностиДокументаСчетФактураПолученный(ТаблицаДанных, СтрокаОписаниеСФ.СчетФактура, Неопределено);
			
		Иначе

			МенеджерДокумента = ОбщегоНазначения.МенеджерОбъектаПоСсылке(СтрокаОснования.ДокументОснование);
			ТаблицаДанных = МенеджерДокумента.ДанныеПолученныхДокументовРегУчет(МассивДокументов, Ложь,
				СтрокаОписаниеСФ.ПФТолькоВРублях, Ошибки);

			УчестьОсобенностиДокументаСчетФактураПолученный(ТаблицаДанных, СтрокаОписаниеСФ.СчетФактура, Неопределено);

		КонецЕсли;

		Если ДанныеДокумента = Неопределено Тогда

			ДанныеДокумента = ТаблицаДанных;

		ИначеЕсли ТипЗнч(ТаблицаДанных) = Тип("ТаблицаЗначений") Тогда

			ОбъединитьСведенияДокументовОснованийСчетовФактур(ДанныеДокумента, ТаблицаДанных);

		КонецЕсли;

	КонецЦикла;

	Возврат ДанныеДокумента;

//		* ПоследняяСтрока - Число - номер последней строки диапазона

Функция ДанныеДокументаОтгрузки(СтрокаДанных = Неопределено)

				Настройки.ДатаОтгрузочногоДокумента, Строка(Настройки.НомерОтгрузочногоДокумента), "");

// Возвращает текст - представление одного документа отгрузки для вывода в строке 5а счета-фактуры.
// В зависимости от переданных параметров формат текста может быть:
//	- № <номер> от <дата>
//	- № п/п <диапазон строк> № <номер> от <дата>
//	- <Наименование>, № <номер> от <дата>
//
// Параметры:
//  ДанныеДокументаОтгрузки	 - Структура - см. описание ДанныеДокументаОтгрузки
//  ПараметрыПредставления	 - Структура - см. Документы.СчетФактура.ОписаниеПечатнойФормыСчетаФактуры
// 
// Возвращаемое значение:
//   - Строка - представление документа
//
Функция ПредставлениеДокументаОтгрузки(ДанныеДокументаОтгрузки, ПараметрыПредставления)

	ВыводитьПункты = ПараметрыПредставления.ВыводитьПунктыДокументаОтгрузки;
	ВыводитьНаименование = ПараметрыПредставления.ВыводитьНаименованиеДокументаОтгрузки;
	
	Если ТипЗнч(ДанныеДокументаОтгрузки.ДатаОтгрузочногоДокумента) = Тип("Дата") Тогда
		ШаблонДатаОтгрузки = НСтр("ru = '%1 г.'");
		ТекстДата = СтрШаблон(ШаблонДатаОтгрузки, Формат(ДанныеДокументаОтгрузки.ДатаОтгрузочногоДокумента, "ДЛФ='Д'"));
	Иначе
		ТекстДата = ДанныеДокументаОтгрузки.ДатаОтгрузочногоДокумента;
	КонецЕсли;
	
	ШаблонНомерДата = НСтр("ru = '№%1 от %2'");
	Представление = СтрШаблон(ШаблонНомерДата, ДанныеДокументаОтгрузки.НомерОтгрузочногоДокумента, ТекстДата);
	
	Если ВыводитьНаименование И ЗначениеЗаполнено(ДанныеДокументаОтгрузки.НаименованиеОтгрузочногоДокумента) Тогда
		Представление = СтрШаблон("%1, %2", ДанныеДокументаОтгрузки.НаименованиеОтгрузочногоДокумента, Представление);
	КонецЕсли;
	
	Если ВыводитьПункты Тогда
		
		Если НЕ ЗначениеЗаполнено(ДанныеДокументаОтгрузки.ПерваяСтрока) Тогда
			ДиапазонСтрок = "";
		ИначеЕсли ДанныеДокументаОтгрузки.ПерваяСтрока = ДанныеДокументаОтгрузки.ПоследняяСтрока Тогда
			ДиапазонСтрок = "" + ДанныеДокументаОтгрузки.ПерваяСтрока;
		Иначе
			ДиапазонСтрок = СтрШаблон("%1-%2", ДанныеДокументаОтгрузки.ПерваяСтрока, ДанныеДокументаОтгрузки.ПоследняяСтрока);
		КонецЕсли;
		
		ШаблонДиапазон = НСтр("ru = '№ п/п %1 %2'");
		Представление = СтрШаблон(ШаблонДиапазон, ДиапазонСтрок, Представление);
		
	КонецЕсли;
	
	Возврат Представление;
	
КонецФункции

// Готовим все количества до вывода строк. При ошибке вся форма сохраняет штатные кг.
Функция УПДЛоткиПодготовитьКарту(ДанныеОбъекта, ОбластьСтроки)
	Попытка
		Карта = Новый Массив;
		КэшДокументов = Новый Соответствие;
		ИспользованныеСтроки = Новый Соответствие;
		КоличествоСтрокКГ = 0;
		Для Каждого СтрокаПечати Из ДанныеОбъекта.ТаблицаЗапасы Цикл
			КоличествоЛотков = Неопределено;
			Если УПДЛоткиСтрокаВКГ(СтрокаПечати) Тогда
				Если ДанныеОбъекта.ВидОперации = Перечисления.ВидыОперацийСчетФактура.Аванс Тогда
					ВызватьИсключение "Для авансового УПД количество лотков не подставляется.";
				КонецЕсли;
				Для Каждого ИмяПризнака Из СтрРазделить("ЭтоНабор,НеобходимоВыделитьКакСоставНабора", ",") Цикл
					Если СтрокаПечати.Владелец().Колонки.Найти(ИмяПризнака) <> Неопределено
						И СтрокаПечати[ИмяПризнака] Тогда
						ВызватьИсключение "Для набора или его комплектующих источник лотков требует отдельного сопоставления.";
					КонецЕсли;
				КонецЦикла;
				КоличествоКГ = СтрокаПечати.Количество;
				Если ТипЗнч(КоличествоКГ) <> Тип("Число") Тогда
					ВызватьИсключение "Количество товара в кг не является числом.";
				КонецЕсли;
				Если ТипЗнч(СтрокаПечати.КоэффициентЕдиницыИзмерения) = Тип("Число") Тогда
					КоличествоКГ = Окр(КоличествоКГ * СтрокаПечати.КоэффициентЕдиницыИзмерения, 3);
				КонецЕсли;
				// Штатная обработка пустых значений выводит прочерк для нулевого количества.
				Если КоличествоКГ <> 0 Тогда
					КоличествоЛотков = УПДЛоткиКоличествоИзСтрокиИсточника(
						СтрокаПечати, КоличествоКГ, КэшДокументов, ИспользованныеСтроки);
					УПДЛоткиПроверитьКоличество(КоличествоЛотков, КоличествоКГ, "Исходная строка УПД");
					КоличествоСтрокКГ = КоличествоСтрокКГ + 1;
				КонецЕсли;
			КонецЕсли;
			Карта.Добавить(КоличествоЛотков);
		КонецЦикла;
		Если КоличествоСтрокКГ = 0 Тогда
			Возврат Неопределено;
		КонецЕсли;
		// Читаем параметр до вывода первого товара: иной макет не меняется частично.
		ПрежнийПараметр = ОбластьСтроки.Параметры.Количество;
		Возврат Карта;
	Исключение
		СообщениеЛотков = Новый СообщениеПользователю;
		СообщениеЛотков.Текст = "СФ/УПД-ЛОТКИ: печатное количество не изменено. " + ОписаниеОшибки();
		СообщениеЛотков.Сообщить();
		Возврат Неопределено;
	КонецПопытки;
КонецФункции

Функция УПДЛоткиСтрокаВКГ(СтрокаПечати)
	Колонки = СтрокаПечати.Владелец().Колонки;
	КодЕдиницы = "";
	НаименованиеЕдиницы = "";
	Если Колонки.Найти("ЕдиницаИзмеренияПоОКЕИ_Код") <> Неопределено Тогда
		КодЕдиницы = СокрЛП(Строка(СтрокаПечати["ЕдиницаИзмеренияПоОКЕИ_Код"]));
	КонецЕсли;
	Если Колонки.Найти("ЕдиницаИзмеренияПоОКЕИ_Наименование") <> Неопределено Тогда
		НаименованиеЕдиницы = УПДЛоткиТекст(Строка(СтрокаПечати["ЕдиницаИзмеренияПоОКЕИ_Наименование"]));
	КонецЕсли;
	Если КодЕдиницы = "166" Тогда
		Возврат Истина;
	ИначеЕсли Не ПустаяСтрока(КодЕдиницы) Тогда
		Если НаименованиеЕдиницы = "кг" Или НаименованиеЕдиницы = "kg" Тогда
			ВызватьИсключение "Наименование кг не совпадает с кодом ОКЕИ строки.";
		КонецЕсли;
		Возврат Ложь;
	КонецЕсли;
	Возврат НаименованиеЕдиницы = "кг" Или НаименованиеЕдиницы = "kg";
КонецФункции


// Сопоставление по исходной ссылке и исходному НомерСтроки DTO, не по печатному НПП.
Функция УПДЛоткиКоличествоИзСтрокиИсточника(СтрокаПечати, КоличествоКГ, КэшДокументов, ИспользованныеСтроки)

	Если ТипЗнч(КоличествоКГ) <> Тип("Число") Тогда
		ВызватьИсключение "Не определено числовое исходное количество строки УПД в кг.";
	КонецЕсли;
	КолонкиИсточника = СтрокаПечати.Владелец().Колонки;
	Для Каждого ИмяКолонки Из СтрРазделить("Ссылка,НомерСтроки,Номенклатура,Характеристика", ",") Цикл
		Если КолонкиИсточника.Найти(ИмяКолонки) = Неопределено Тогда
			ВызватьИсключение "В данных УПД нет колонки " + ИмяКолонки + ": связь с сохранённой строкой накладной не подтверждена.";
		КонецЕсли;
	КонецЦикла;
	СсылкаИсточника = СтрокаПечати["Ссылка"];
	Если ТипЗнч(СсылкаИсточника) <> Тип("ДокументСсылка.РасходнаяНакладная") Тогда
		ВызватьИсключение "Источник строки УПД не является расходной накладной. Для сгруппированных и иных документов количество сохранено.";
	КонецЕсли;
	Если Не ЗначениеЗаполнено(СсылкаИсточника) Тогда
		ВызватьИсключение "В строке УПД не заполнена ссылка исходной расходной накладной.";
	КонецЕсли;
	НомерИсточника = УПДЛоткиЧисловоеЗначение(СтрокаПечати["НомерСтроки"]);
	Если НомерИсточника = Неопределено Тогда
		ВызватьИсключение "Не прочитан исходный номер строки расходной накладной.";
	КонецЕсли;
	Если НомерИсточника <= 0 Или НомерИсточника <> Цел(НомерИсточника) Тогда
		ВызватьИсключение "В данных УПД исходный номер строки накладной должен быть положительным целым.";
	КонецЕсли;

	ОбъектИсточника = КэшДокументов.Получить(СсылкаИсточника);
	Если ОбъектИсточника = Неопределено Тогда
		ОбъектИсточника = СсылкаИсточника.ПолучитьОбъект();
		Если ОбъектИсточника = Неопределено Тогда
			ВызватьИсключение "Не удалось прочитать сохранённую расходную накладную — источник УПД.";
		КонецЕсли;
		КэшДокументов.Вставить(СсылкаИсточника, ОбъектИсточника);
	КонецЕсли;
	Если ОбъектИсточника.ПометкаУдаления Тогда
		ВызватьИсключение "Исходная расходная накладная помечена на удаление: её лотки не используются.";
	КонецЕсли;
	МетаЗапасы = ОбъектИсточника.Метаданные().ТабличныеЧасти.Найти("Запасы");
	УПДЛоткиПроверитьПоляСопоставления(МетаЗапасы);
	СохраненнаяСтрока = Неопределено;
	Для Каждого СтрокаИсточника Из ОбъектИсточника.Запасы Цикл
		// НомерСтроки — стандартное свойство строки табличной части.
		Если СтрокаИсточника.НомерСтроки = НомерИсточника Тогда
			Если СохраненнаяСтрока <> Неопределено Тогда
				ВызватьИсключение "В исходной накладной обнаружено несколько строк с одним исходным номером.";
			КонецЕсли;
			СохраненнаяСтрока = СтрокаИсточника;
		КонецЕсли;
	КонецЦикла;
	Если СохраненнаяСтрока = Неопределено Тогда
		ВызватьИсключение "В исходной накладной отсутствует сохранённая строка " + Строка(НомерИсточника) + ".";
	КонецЕсли;
	Если СохраненнаяСтрока["Номенклатура"] <> СтрокаПечати["Номенклатура"]
		Или СохраненнаяСтрока["Характеристика"] <> СтрокаПечати["Характеристика"] Тогда
		ВызватьИсключение "Номенклатура или характеристика УПД не совпадает с исходной строкой накладной.";
	КонецЕсли;
	Если КолонкиИсточника.Найти("Код") <> Неопределено Тогда
		Если УПДЛоткиТекст(Строка(СтрокаПечати["Код"]))
			<> УПДЛоткиТекст(Строка(СохраненнаяСтрока.Номенклатура.Код)) Тогда
			ВызватьИсключение "Код номенклатуры УПД не совпадает с сохранённой строкой накладной.";
		КонецЕсли;
	КонецЕсли;
	КилограммыИсточника = УПДЛоткиКгСохраненнойСтроки(СохраненнаяСтрока);
	Если КилограммыИсточника = Неопределено Тогда
		УПДЛоткиДиагностикаОбъекта(ОбъектИсточника, СохраненнаяСтрока, "Источник УПД");
		ВызватьИсключение "В исходной накладной не подтверждены базовые кг или коэффициент единицы измерения.";
	КонецЕсли;
	РазницаКоличество = КилограммыИсточника - КоличествоКГ;
	Если РазницаКоличество > 0.0005 Или РазницаКоличество < -0.0005 Тогда
		УПДЛоткиДиагностикаОбъекта(ОбъектИсточника, СохраненнаяСтрока, "Источник УПД");
		ВызватьИсключение "Исходные кг УПД не совпадают с сохранённой строкой накладной. Свёрнутые и частичные количества не подменяются лотками.";
	КонецЕсли;

	ИспользованныеНомера = ИспользованныеСтроки.Получить(СсылкаИсточника);
	Если ИспользованныеНомера <> Неопределено Тогда
		Если ИспользованныеНомера.Получить(НомерИсточника) <> Неопределено Тогда
			ВызватьИсключение "Одна исходная строка накладной повторно включена в УПД: повторный учёт лотков остановлен.";
		КонецЕсли;
	КонецЕсли;
	КоличествоЛотков = УПДЛоткиПолучитьКоличество(СохраненнаяСтрока, МетаЗапасы, КоличествоКГ, Ложь);
	Если КоличествоЛотков = Неопределено Тогда
		УПДЛоткиДиагностикаОбъекта(ОбъектИсточника, СохраненнаяСтрока, "Источник УПД");
		ВызватьИсключение "Исходная строка накладной найдена, но её количество лотков не подтверждено. Количество УПД сохранено.";
	КонецЕсли;
	УПДЛоткиПроверитьКоличество(КоличествоЛотков, КоличествоКГ, "Сохранённая строка источника УПД");
	Если ИспользованныеНомера = Неопределено Тогда
		ИспользованныеНомера = Новый Соответствие;
		ИспользованныеСтроки.Вставить(СсылкаИсточника, ИспользованныеНомера);
	КонецЕсли;
	ИспользованныеНомера.Вставить(НомерИсточника, Истина);
	Возврат КоличествоЛотков;

КонецФункции


// Только подтверждённые реквизиты строки или введённое количество упаковок-лотков.
// Общие УПДЛоткиТекст и УПДЛоткиЧисло добавляются основным помощником печати.
Функция УПДЛоткиПолучитьКоличество(ИсходнаяСтрокаЗапасов, МетаЗапасы, КоличествоКГ,
	ВызыватьОшибкуПриОтсутствии = Истина)

	Если ТипЗнч(КоличествоКГ) <> Тип("Число") Тогда
		ВызватьИсключение "Не определено числовое количество товара в кг.";
	КонецЕсли;
	ПолеЛотков = УПДЛоткиРеквизитПоИменам(МетаЗапасы,
		"количестволотков,колволотков,коллотков");
	Если ПолеЛотков <> Неопределено Тогда
		КоличествоЛотков = УПДЛоткиЧисловоеЗначение(ИсходнаяСтрокаЗапасов[ПолеЛотков.Имя]);
		УПДЛоткиПроверитьКоличество(КоличествоЛотков, КоличествоКГ, ПолеЛотков.Имя);
		Возврат КоличествоЛотков;
	КонецЕсли;

	Если МетаЗапасы.Реквизиты.Найти("ЕдиницаИзмерения") <> Неопределено
		И МетаЗапасы.Реквизиты.Найти("Количество") <> Неопределено Тогда
		Единица = ИсходнаяСтрокаЗапасов["ЕдиницаИзмерения"];
		Если ТипЗнч(Единица) = Тип("СправочникСсылка.ЕдиницыИзмерения")
			И ЗначениеЗаполнено(Единица) Тогда
			// Как в рабочей ТОРГ-12: КоличествоМест = введённое количество упаковок,
			// КоличествоВОдномМесте = коэффициент пользовательской единицы.
			Если МетаЗапасы.Реквизиты.Найти("Номенклатура") <> Неопределено
				И УПДЛоткиБазоваяЕдиницаКГ(ИсходнаяСтрокаЗапасов["Номенклатура"])
				И Единица.Метаданные().Реквизиты.Найти("Коэффициент") <> Неопределено Тогда
				КоличествоЛотков = УПДЛоткиЧисловоеЗначение(ИсходнаяСтрокаЗапасов["Количество"]);
				КоэффициентУпаковки = Единица.Коэффициент;
				Если ТипЗнч(КоэффициентУпаковки) <> Тип("Число") Или КоэффициентУпаковки <= 0 Тогда
					ВызватьИсключение "У упаковки товара в кг не задан положительный коэффициент.";
				КонецЕсли;
				ИмяУпаковки = УПДЛоткиТекст(Строка(Единица));
				ЭтоУпаковка = СтрНайти(ИмяУпаковки, "лоток") > 0 Или СтрНайти(ИмяУпаковки, "лотк") > 0
					Или СтрНайти(ИмяУпаковки, "упаков") > 0;
				Если Единица.Метаданные().Реквизиты.Найти("ЕдиницаИзмеренияПоКлассификатору") <> Неопределено Тогда
					КлассификаторУпаковки = Единица.ЕдиницаИзмеренияПоКлассификатору;
					Если ТипЗнч(КлассификаторУпаковки) = Тип("СправочникСсылка.КлассификаторЕдиницИзмерения")
						И ЗначениеЗаполнено(КлассификаторУпаковки)
						И СокрЛП(Строка(КлассификаторУпаковки.Код)) = "796" Тогда
						ЭтоУпаковка = Истина;
					КонецЕсли;
				КонецЕсли;
				// Коэффициент перевода единиц сам по себе не доказывает лотки (например, г → кг).
				Если ЭтоУпаковка Тогда
					УПДЛоткиПроверитьКоличество(КоличествоЛотков, КоличествоКГ, "Количество упаковок-лотков");
					РазницаКГ = КоличествоЛотков * КоэффициентУпаковки - КоличествоКГ;
					Если РазницаКГ <= 0.0005 И РазницаКГ >= -0.0005 Тогда
						Возврат КоличествоЛотков;
					КонецЕсли;
					ВызватьИсключение "Количество упаковок, умноженное на коэффициент, не совпадает с напечатанными кг.";
				КонецЕсли;
			КонецЕсли;
		КонецЕсли;
	КонецЕсли;

	ПолеВеса = УПДЛоткиРеквизитПоИменам(МетаЗапасы, "веслотка,массалотка");
	Если ПолеВеса <> Неопределено Тогда
		ВесЛотка = УПДЛоткиЧисловоеЗначение(ИсходнаяСтрокаЗапасов[ПолеВеса.Имя]);
		Если ВесЛотка = Неопределено Тогда
			ВызватьИсключение "Не прочитан вес лотка из реквизита " + ПолеВеса.Имя + ".";
		КонецЕсли;
		Если ВесЛотка <= 0 Тогда
			ВызватьИсключение "Для расчёта лотков реквизит " + ПолеВеса.Имя + " должен быть больше нуля и содержать кг на лоток.";
		КонецЕсли;
		Возврат КоличествоКГ / ВесЛотка;
	КонецЕсли;

	Если Не ВызыватьОшибкуПриОтсутствии Тогда
		Возврат Неопределено;
	КонецЕсли;
	ВызватьИсключение "В строке документа нет подтверждённого количества лотков: нужны количество упаковок с коэффициентом в кг, реквизит КоличествоЛотков или ВесЛотка в кг. Печатное количество не изменено.";

КонецФункции

Функция УПДЛоткиБазоваяЕдиницаКГ(Номенклатура)
	Если ТипЗнч(Номенклатура) <> Тип("СправочникСсылка.Номенклатура") Или Не ЗначениеЗаполнено(Номенклатура) Тогда
		Возврат Ложь;
	КонецЕсли;
	Если Номенклатура.Метаданные().Реквизиты.Найти("ЕдиницаИзмерения") = Неопределено Тогда
		Возврат Ложь;
	КонецЕсли;
	БазоваяЕдиница = Номенклатура.ЕдиницаИзмерения;
	Возврат ТипЗнч(БазоваяЕдиница) = Тип("СправочникСсылка.КлассификаторЕдиницИзмерения")
		И ЗначениеЗаполнено(БазоваяЕдиница) И СокрЛП(Строка(БазоваяЕдиница.Код)) = "166";
КонецФункции

// Учитываем только точное имя или синоним: плановые счётчики не подходят.
// Несколько подходящих реквизитов запрещены: источник нельзя выбирать наугад.
Функция УПДЛоткиРеквизитПоИменам(МетаЗапасы, ИменаЧерезЗапятую)

	Результат = Неопределено;
	ДопустимыеИмена = СтрРазделить(ИменаЧерезЗапятую, ",");
	Для Каждого Реквизит Из МетаЗапасы.Реквизиты Цикл
		Если Не Реквизит.Тип.СодержитТип(Тип("Число"))
			И Не Реквизит.Тип.СодержитТип(Тип("Строка")) Тогда
			Продолжить;
		КонецЕсли;
		Имя = УПДЛоткиИмяПоля(Реквизит.Имя);
		Синоним = УПДЛоткиИмяПоля(Реквизит.Синоним);
		Подходит = Ложь;
		Для Каждого РазрешенноеИмя Из ДопустимыеИмена Цикл
			Если Имя = РазрешенноеИмя
				Или Синоним = РазрешенноеИмя Тогда
				Подходит = Истина;
				Прервать;
			КонецЕсли;
		КонецЦикла;
		Если Подходит Тогда
			Если Результат <> Неопределено Тогда
				ВызватьИсключение "Найдено несколько возможных реквизитов лотков: "
					+ Результат.Имя + ", " + Реквизит.Имя + ". Нужен однозначный источник.";
			КонецЕсли;
			Результат = Реквизит;
		КонецЕсли;
	КонецЦикла;
	Возврат Результат;

КонецФункции

Функция УПДЛоткиИмяПоля(Имя)
	Результат = УПДЛоткиТекст(Имя);
	Результат = СтрЗаменить(Результат, "_", "");
	Результат = СтрЗаменить(Результат, "-", "");
	Возврат СтрЗаменить(Результат, ".", "");
КонецФункции

Функция УПДЛоткиЧисловоеЗначение(Значение)
	Если ТипЗнч(Значение) = Тип("Число") Тогда
		Возврат Значение;
	ИначеЕсли ТипЗнч(Значение) = Тип("Строка") Тогда
		Возврат УПДЛоткиЧисло(Значение);
	КонецЕсли;
	Возврат Неопределено;
КонецФункции

Процедура УПДЛоткиПроверитьКоличество(КоличествоЛотков, КоличествоКГ, ИмяИсточника)
	Если КоличествоЛотков = Неопределено Тогда
		ВызватьИсключение "Не прочитано количество лотков из " + ИмяИсточника + ".";
	КонецЕсли;
	Если КоличествоКГ <> 0 И КоличествоЛотков = 0 Тогда
		ВызватьИсключение "В " + ИмяИсточника + " указано ноль лотков при непустой строке в кг.";
	КонецЕсли;
	Если (КоличествоКГ > 0 И КоличествоЛотков < 0)
		Или (КоличествоКГ < 0 И КоличествоЛотков > 0) Тогда
		ВызватьИсключение "Знак количества лотков из " + ИмяИсточника + " отличается от количества в кг.";
	КонецЕсли;
КонецПроцедуры


Процедура УПДЛоткиПроверитьПоляСопоставления(МетаЗапасы)
	Если МетаЗапасы = Неопределено Тогда
		ВызватьИсключение "В связанном документе не найдена табличная часть Запасы.";
	КонецЕсли;
	Для Каждого ИмяПоля Из СтрРазделить("Номенклатура,Характеристика,ЕдиницаИзмерения,Количество", ",") Цикл
		Если МетаЗапасы.Реквизиты.Найти(ИмяПоля) = Неопределено Тогда
			ВызватьИсключение "Для сопоставления строк не найден реквизит Запасы." + ИмяПоля + ".";
		КонецЕсли;
	КонецЦикла;
КонецПроцедуры

Функция УПДЛоткиКгСохраненнойСтроки(СтрокаЗапасов)
	Количество = УПДЛоткиЧисловоеЗначение(СтрокаЗапасов["Количество"]);
	Если Количество = Неопределено Тогда
		Возврат Неопределено;
	КонецЕсли;
	Единица = СтрокаЗапасов["ЕдиницаИзмерения"];
	Если ТипЗнч(Единица) = Тип("СправочникСсылка.КлассификаторЕдиницИзмерения") Тогда
		Если ЗначениеЗаполнено(Единица) И СокрЛП(Строка(Единица.Код)) = "166" Тогда
			Возврат Количество;
		КонецЕсли;
	ИначеЕсли ТипЗнч(Единица) = Тип("СправочникСсылка.ЕдиницыИзмерения") Тогда
		Если Не ЗначениеЗаполнено(Единица) Или Не УПДЛоткиБазоваяЕдиницаКГ(СтрокаЗапасов["Номенклатура"]) Тогда
			Возврат Неопределено;
		КонецЕсли;
		Если Единица.Метаданные().Реквизиты.Найти("Коэффициент") = Неопределено Тогда
			Возврат Неопределено;
		КонецЕсли;
		Коэффициент = УПДЛоткиЧисловоеЗначение(Единица.Коэффициент);
		Если Коэффициент = Неопределено Тогда
			Возврат Неопределено;
		КонецЕсли;
		Если Коэффициент > 0 Тогда
			Возврат Количество * Коэффициент;
		КонецЕсли;
	КонецЕсли;
	Возврат Неопределено;
КонецФункции


// Диагностика только читает имена полей и единицы. Документ не записывается.
Процедура УПДЛоткиДиагностикаОбъекта(ОбъектДокумента, СтрокаЗапасов, ПодписьИсточника)
	Попытка
		МетаОбъекта = ОбъектДокумента.Метаданные();
		ТекстДиагностики = "СФ/УПД-ДИАГ: " + ПодписьИсточника + "; документ=" + МетаОбъекта.Имя;
		МетаЗапасы = МетаОбъекта.ТабличныеЧасти.Найти("Запасы");
		Если МетаЗапасы = Неопределено Тогда
			ТекстДиагностики = ТекстДиагностики + Символы.ПС + "Табличной части Запасы нет.";
		Иначе
			Если СтрокаЗапасов <> Неопределено Тогда
				ТекстДиагностики = ТекстДиагностики + "; строка=" + Строка(СтрокаЗапасов.НомерСтроки);
				Если МетаЗапасы.Реквизиты.Найти("Количество") <> Неопределено Тогда
					КоличествоСтроки = СтрокаЗапасов["Количество"];
					Если ТипЗнч(КоличествоСтроки) = Тип("Число") Тогда
						ТекстДиагностики = ТекстДиагностики + "; введённое количество=" + Строка(КоличествоСтроки);
					КонецЕсли;
				КонецЕсли;
				Если МетаЗапасы.Реквизиты.Найти("ЕдиницаИзмерения") <> Неопределено Тогда
					ЕдиницаСтроки = СтрокаЗапасов["ЕдиницаИзмерения"];
					ТекстДиагностики = ТекстДиагностики + Символы.ПС
						+ "Единица: тип=" + Строка(ТипЗнч(ЕдиницаСтроки));
					Если ТипЗнч(ЕдиницаСтроки) = Тип("СправочникСсылка.ЕдиницыИзмерения")
						И ЗначениеЗаполнено(ЕдиницаСтроки) Тогда
						ТекстДиагностики = ТекстДиагностики + "; наименование=" + ЕдиницаСтроки.Наименование;
						Если ЕдиницаСтроки.Метаданные().Реквизиты.Найти("Коэффициент") <> Неопределено Тогда
							ТекстДиагностики = ТекстДиагностики + "; коэффициент=" + Строка(ЕдиницаСтроки.Коэффициент);
						КонецЕсли;
					ИначеЕсли ТипЗнч(ЕдиницаСтроки) = Тип("СправочникСсылка.КлассификаторЕдиницИзмерения")
						И ЗначениеЗаполнено(ЕдиницаСтроки) Тогда
						ТекстДиагностики = ТекстДиагностики + "; ОКЕИ=" + СокрЛП(Строка(ЕдиницаСтроки.Код));
					КонецЕсли;
				КонецЕсли;
				Если МетаЗапасы.Реквизиты.Найти("Номенклатура") <> Неопределено Тогда
					ТекстДиагностики = ТекстДиагностики + Символы.ПС + "Базовая единица номенклатуры кг="
						+ ?(УПДЛоткиБазоваяЕдиницаКГ(СтрокаЗапасов["Номенклатура"]), "Да", "Нет");
				КонецЕсли;
			КонецЕсли;
			ТекстДиагностики = ТекстДиагностики + Символы.ПС + "Числовые/строковые поля Запасы (имя | синоним):";
			Для Каждого МетаПоля Из МетаЗапасы.Реквизиты Цикл
				Если МетаПоля.Тип.СодержитТип(Тип("Число")) Или МетаПоля.Тип.СодержитТип(Тип("Строка")) Тогда
					ТекстДиагностики = ТекстДиагностики + Символы.ПС + МетаПоля.Имя + " | " + МетаПоля.Синоним;
				КонецЕсли;
			КонецЦикла;
		КонецЕсли;
		Для Каждого ИмяСвязи Из СтрРазделить("ДокументОснование,ОснованиеПечатиСсылка", ",") Цикл
			Если МетаОбъекта.Реквизиты.Найти(ИмяСвязи) <> Неопределено Тогда
				СсылкаСвязи = ОбъектДокумента[ИмяСвязи];
				ТекстДиагностики = ТекстДиагностики + Символы.ПС + ИмяСвязи
					+ ": тип=" + Строка(ТипЗнч(СсылкаСвязи))
					+ "; заполнено=" + ?(ЗначениеЗаполнено(СсылкаСвязи), "Да", "Нет");
			КонецЕсли;
		КонецЦикла;
		СообщениеДиагностики = Новый СообщениеПользователю;
		СообщениеДиагностики.Текст = ТекстДиагностики;
		СообщениеДиагностики.Сообщить();
	Исключение
		// Ошибка диагностики не заменяет первоначальную причину остановки правки.
	КонецПопытки;
КонецПроцедуры


Функция УПДЛоткиТекст(ТекстУПД)
	РезультатУПД = НРег(СокрЛП(ТекстУПД));
	РезультатУПД = СтрЗаменить(РезультатУПД, " ", "");
	РезультатУПД = СтрЗаменить(РезультатУПД, Символ(160), "");
	РезультатУПД = СтрЗаменить(РезультатУПД, Символ(8239), "");
	РезультатУПД = СтрЗаменить(РезультатУПД, Символ(8201), "");
	РезультатУПД = СтрЗаменить(РезультатУПД, Символы.ПС, "");
	РезультатУПД = СтрЗаменить(РезультатУПД, Символы.ВК, "");
	Возврат РезультатУПД;
КонецФункции

// Читаем цифры, запятую/точку без зависимости от регионального разделителя.
Функция УПДЛоткиЧисло(ТекстУПД)
	ТекстЧислаУПД = УПДЛоткиТекст(ТекстУПД);
	Если ПустаяСтрока(ТекстЧислаУПД) Тогда
		Возврат Неопределено;
	КонецЕсли;
	ЗнакУПД = 1;
	Если Лев(ТекстЧислаУПД, 1) = "-" Тогда
		ЗнакУПД = -1;
		ТекстЧислаУПД = Сред(ТекстЧислаУПД, 2);
	ИначеЕсли Лев(ТекстЧислаУПД, 1) = "+" Тогда
		ТекстЧислаУПД = Сред(ТекстЧислаУПД, 2);
	КонецЕсли;
	ЗначениеУПД = 0;
	ДелительУПД = 1;
	ЕстьЦифраУПД = Ложь;
	ПослеРазделителяУПД = Ложь;
	Для ПозицияУПД = 1 По СтрДлина(ТекстЧислаУПД) Цикл
		СимволУПД = Сред(ТекстЧислаУПД, ПозицияУПД, 1);
		Если СимволУПД = "," Или СимволУПД = "." Тогда
			Если ПослеРазделителяУПД Тогда
				Возврат Неопределено;
			КонецЕсли;
			ПослеРазделителяУПД = Истина;
		Иначе
			ЦифраУПД = СтрНайти("0123456789", СимволУПД) - 1;
			Если ЦифраУПД < 0 Тогда
				Возврат Неопределено;
			КонецЕсли;
			ЕстьЦифраУПД = Истина;
			ЗначениеУПД = ЗначениеУПД * 10 + ЦифраУПД;
			Если ПослеРазделителяУПД Тогда
				ДелительУПД = ДелительУПД * 10;
			КонецЕсли;
		КонецЕсли;
	КонецЦикла;
	Если Не ЕстьЦифраУПД Тогда
		Возврат Неопределено;
	КонецЕсли;
	Возврат ЗнакУПД * ЗначениеУПД / ДелительУПД;
КонецФункции
